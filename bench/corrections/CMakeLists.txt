foreach(mode IN ITEMS transform plain position both)
    set(POSE_TYPE Pose)
    set(POSITION_TAG "")
    set(ROTATION_TAG "")
    if(mode STREQUAL "position" OR mode STREQUAL "both")
        set(POSITION_TAG "[Position]")
    endif()
    if(mode STREQUAL "both")
        set(ROTATION_TAG "[Rotation]")
    endif()
    set(POSE_DECL "component Pose { ${POSITION_TAG} float3 localPosition; ${ROTATION_TAG} quaternion localRotation = quaternion.identity; float3 localScale = float3(1); }")
    if(mode STREQUAL "transform")
        set(POSE_TYPE Transform)
        set(POSE_DECL "")
    endif()
    configure_file(game.tide.in "${CMAKE_CURRENT_BINARY_DIR}/${mode}/game.tide" @ONLY)
    tide_add_game(bench_corrections_${mode} NAME game
        SOURCES "${CMAKE_CURRENT_BINARY_DIR}/${mode}/game.tide" HOST main.c)
    target_compile_definitions(bench_corrections_${mode} PRIVATE
        BENCH_MODE="${mode}" BENCH_TRANSFORM=$<STREQUAL:${mode},transform>)
    add_test(NAME bench_corrections_${mode} COMMAND bench_corrections_${mode} --quick)
    add_test(NAME bench_corrections_${mode}_large COMMAND bench_corrections_${mode} --quick --count 10000)
endforeach()
