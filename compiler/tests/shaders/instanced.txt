// Place and Tint on Corner: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
};
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec3 tide_a1;
layout(location = 2) in vec4 tide_n0;
layout(location = 3) in vec4 tide_n1;
layout(location = 4) in vec4 tide_n2;
layout(location = 5) in vec4 tide_n3;
layout(location = 6) in vec4 tide_n4;
layout(location = 7) in vec4 tide_n5;
layout(location = 8) in ivec4 tide_j0;
out vec3 tide_s1;
flat out vec4 tide_f0;
flat out vec4 tide_f1;
flat out ivec4 tide_g0;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
};
void ts_Place(inout tt_Corner t_vertex) {
    t_vertex.m_position = (t_vertex.m_position + tide_n4.xyz);
    t_vertex.m_normal = (mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(t_vertex.m_normal, 0.0)).xyz;
}

void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1);
    ts_Place(tide_vertex);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_normal;
    tide_f0 = vec4(tide_n4.w, tide_n5.x, tide_n5.y, tide_n5.z);
    tide_f1 = vec4(tide_n5.w, 0.0, 0.0, 0.0);
    tide_g0 = ivec4(tide_j0.x, 0, 0, 0);
}

// Place and Tint on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
};
in vec3 tide_s1;
flat in vec4 tide_f0;
flat in vec4 tide_f1;
flat in ivec4 tide_g0;
out vec4 tide_pixel;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
};
struct tt_Surface {
    vec4 m_tint;
    float m_gain;
};
float tide_f(float x) {
    return x;
}
bool tide_nan(float x) {
    return (floatBitsToInt(x) & 0x7fffffff) > 0x7f800000;
}
float tide_max1(float a, float b) {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
void ts_Tint(tt_Corner t_vertex, inout vec4 t_color) {
    float t_gain = tide_max1(dot(t_vertex.m_normal, (-tide_u0.xyz)), tide_f(0.1));
    t_color = (tt_Surface(tide_f0, tide_f1.x).m_tint * (tt_Surface(tide_f0, tide_f1.x).m_gain * t_gain));
    if ((tide_g0.x == 0)) {
        t_color.a = 0.5;
    }
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Tint(tt_Corner(vec3(0.0), tide_s1), tide_color);
    tide_color = tide_color * tide_tint;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb * tide_color.a + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Place and Tint on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_n4: vec4f;
var<private> tide_n5: vec4f;
var<private> tide_j0: vec4i;
var<private> tide_f0: vec4f;
var<private> tide_f1: vec4f;
var<private> tide_g0: vec4i;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec3f,
    @location(1) @interpolate(flat) tide_f0: vec4f,
    @location(2) @interpolate(flat) tide_f1: vec4f,
    @location(3) @interpolate(flat) tide_g0: vec4i,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
}
struct tt_Surface {
    m_tint: vec4f,
    m_gain: f32,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn ts_Place(t_vertex: ptr<function, tt_Corner>) {
    (*t_vertex).m_position = ((*t_vertex).m_position + tide_n4.xyz);
    (*t_vertex).m_normal = (mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f((*t_vertex).m_normal, 0.0)).xyz;
}

fn ts_Tint(t_vertex: tt_Corner, t_color: ptr<function, vec4f>) {
    var t_gain: f32 = tide_max1(dot(t_vertex.m_normal, (-tide_block.tide_u0.xyz)), tide_f(0.1));
    (*t_color) = (tt_Surface(tide_f0, tide_f1.x).m_tint * (tt_Surface(tide_f0, tide_f1.x).m_gain * t_gain));
    if ((tide_g0.x == 0)) {
        (*t_color).a = 0.5;
    }
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f, @location(6) tide_in4: vec4f, @location(7) tide_in5: vec4f, @location(8) tide_ij0: vec4i) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_n4 = tide_in4;
    tide_n5 = tide_in5;
    tide_j0 = tide_ij0;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    ts_Place(&tide_vertex);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_normal;
    out.tide_f0 = vec4f(tide_n4.w, tide_n5.x, tide_n5.y, tide_n5.z);
    out.tide_f1 = vec4f(tide_n5.w, 0.0, 0.0, 0.0);
    out.tide_g0 = vec4i(tide_j0.x, 0, 0, 0);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_f0 = in.tide_f0;
    tide_f1 = in.tide_f1;
    tide_g0 = in.tide_g0;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Tint(tt_Corner(vec3f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb * tide_color.a + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
