// Defaults on Vertex: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
layout(location = 0) in vec2 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_a2;
struct tt_Vertex {
    vec2 m_position;
    vec2 m_uv;
    vec4 m_color;
};
void main() {
    tt_Vertex tide_vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    vec4 tide_place = vec4(tide_vertex.m_position, 0.0, 1.0);
    gl_Position = tide_camera * tide_place;
}

// Defaults on Vertex: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
out vec4 tide_pixel;
struct tt_Vertex {
    vec2 m_position;
    vec2 m_uv;
    vec4 m_color;
};
struct tt_BoneTransform {
    vec3 m_position;
    vec4 m_rotation;
    vec3 m_scale;
};
void ts_Defaults(tt_Vertex t_corner, inout vec4 t_color) {
    tt_BoneTransform t_literal = tt_BoneTransform(vec3(0.0), vec4(0.0, 0.0, 0.0, 1.0), vec3(1.0));
    tt_BoneTransform t_value = tt_BoneTransform(vec3(0.0), vec4(0.0, 0.0, 0.0, 1.0), vec3(1.0));
    t_color = vec4(t_literal.m_scale.x, t_value.m_rotation.w, t_literal.m_position.x, 1.0);
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Defaults(tt_Vertex(vec2(0.0), vec2(0.0), vec4(0.0)), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Defaults on Vertex: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
struct tt_BoneTransform {
    m_position: vec3f,
    m_rotation: vec4f,
    m_scale: vec3f,
}
fn ts_Defaults(t_corner: tt_Vertex, t_color: ptr<function, vec4f>) {
    var t_literal: tt_BoneTransform = tt_BoneTransform(vec3f(), vec4f(0.0, 0.0, 0.0, 1.0), vec3f(1.0));
    var t_value: tt_BoneTransform = tt_BoneTransform(vec3f(), vec4f(0.0, 0.0, 0.0, 1.0), vec3f(1.0));
    (*t_color) = vec4f(t_literal.m_scale.x, t_value.m_rotation.w, t_literal.m_position.x, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Defaults(tt_Vertex(vec2f(), vec2f(), vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
