fn tide_min1(a: f32, b: f32) -> f32 {
    return select(select(select(b, a, a < b), b, (bitcast<i32>(a) & 0x7fffffff) > 0x7f800000), a, (bitcast<i32>(b) & 0x7fffffff) > 0x7f800000);
}
fn tide_max1(a: f32, b: f32) -> f32 {
    return select(select(select(b, a, a > b), b, (bitcast<i32>(a) & 0x7fffffff) > 0x7f800000), a, (bitcast<i32>(b) & 0x7fffffff) > 0x7f800000);
}
fn tide_clamp1(a: f32, b: f32, c: f32) -> f32 {
    return select(tide_max1(b, tide_min1(c, a)), b, (bitcast<i32>(a) & 0x7fffffff) > 0x7f800000);
}

    var tide_choice0: vec2f = t_vertex.m_uv;
    var tide_choice1: i32 = 1;
    var t_c: vec4f = textureSampleLevel(pixels2, sampler1, tide_choice0, tide_clamp1(1.0, 0.0, 31.0));
    var t_d: i32 = (tide_number3_0_5(t_vertex.m_uv) + tide_number4_1_5(t_vertex.m_uv));
    (*t_color) = vec4f((t_a.x + t_c.y), f32(t_d), f32(tide_block.tide_v0.xy.x), 1.0);
}
