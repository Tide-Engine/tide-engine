    vec3 m_position;

    vec2 m_uv;
};
struct tt_FromLamp {
    vec3 m_place;
    float m_glow;
    vec4 m_tint;
};
struct tt_Fog {
    float m_depth;
    vec2 m_drift;
};
float tide_f(float x) {
    return x;
}
bool tide_nan(float x) {
    return (floatBitsToInt(x) & 0x7fffffff) > 0x7f800000;
}
float tide_max1(float a, float b) {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
vec3 tide_normalize3(vec3 v) {
    float l = dot(v, v);
    if (l > 1.17549435e-38) { return v * inversesqrt(l); }
    return vec3(0.0);
}
void tf_Light(vec3 t_from, vec3 t_normal, inout float t_glow) {
    t_glow = tide_max1(dot(t_normal, tide_normalize3((-t_from))), tide_f(0.0));
}

void ts_Place(inout tt_Corner t_corner, inout tt_FromLamp t_lamp, inout tt_Fog t_fog) {
    t_corner.m_position.y = (t_corner.m_position.y + 0.1);
    t_lamp.m_place = (t_corner.m_position - tide_u0.xyz);
    tf_Light(t_lamp.m_place, t_corner.m_normal, t_lamp.m_glow);
    t_lamp.m_tint = vec4(1.0, 0.9, 0.8, 1.0);
    t_fog = tt_Fog(t_corner.m_position.z, t_corner.m_uv);
}

void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1, tide_a2);
    tt_FromLamp tide_handed0 = tt_FromLamp(vec3(0.0), 0.0, vec4(0.0));
    tt_Fog tide_handed1 = tt_Fog(0.0, vec2(0.0));
    ts_Place(tide_vertex, tide_handed0, tide_handed1);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);

    vec2 m_uv;

void ts_Look(tt_Corner t_corner, tt_FromLamp t_lamp, inout vec4 t_color) {
    if ((t_lamp.m_glow <= 0.0)) {
        discard;
    }
    t_color = (t_lamp.m_tint * t_lamp.m_glow);
    t_color.a = t_corner.m_uv.x;
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Look(tt_Corner(vec3(0.0), vec3(0.0), tide_s2), tt_FromLamp(vec3(0.0), tide_h0_1, tide_h0_2), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

    m_position: vec3f,
    m_normal: vec3f,
    m_uv: vec2f,

    m_place: vec3f,
    m_glow: f32,
    m_tint: vec4f,
}
struct tt_Fog {
    m_depth: f32,
    m_drift: vec2f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tf_Light(t_from: vec3f, t_normal: vec3f, t_glow: ptr<function, f32>) {
    (*t_glow) = tide_max1(dot(t_normal, tide_normalize3((-t_from))), tide_f(0.0));
}

fn ts_Place(t_corner: ptr<function, tt_Corner>, t_lamp: ptr<function, tt_FromLamp>, t_fog: ptr<function, tt_Fog>) {
    (*t_corner).m_position.y = ((*t_corner).m_position.y + 0.1);
    (*t_lamp).m_place = ((*t_corner).m_position - tide_block.tide_u0.xyz);
    {
        var t_tide_0copy2: f32 = (*t_lamp).m_glow;
        tf_Light((*t_lamp).m_place, (*t_corner).m_normal, &t_tide_0copy2);
        (*t_lamp).m_glow = t_tide_0copy2;
    }
    (*t_lamp).m_tint = vec4f(1.0, 0.9, 0.8, 1.0);
    (*t_fog) = tt_Fog((*t_corner).m_position.z, (*t_corner).m_uv);
}

    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1, tide_a2);

    var tide_handed1: tt_Fog = tt_Fog(f32(), vec2f());

    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);

}
