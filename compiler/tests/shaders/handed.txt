// Place and Look on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s2: vec2f,
    @location(1) tide_h0_1: f32,
    @location(2) tide_h0_2: vec4f,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
    m_uv: vec2f,
}
struct tt_FromLamp {
    m_place: vec3f,
    m_glow: f32,
    m_tint: vec4f,
}
struct tt_Fog {
    m_depth: f32,
    m_drift: vec2f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tf_Light(t_from: vec3f, t_normal: vec3f, t_glow: ptr<function, f32>) {
    (*t_glow) = tide_max1(dot(t_normal, tide_normalize3((-t_from))), tide_f(0.0));
}

fn ts_Place(t_corner: ptr<function, tt_Corner>, t_lamp: ptr<function, tt_FromLamp>, t_fog: ptr<function, tt_Fog>) {
    (*t_corner).m_position.y = ((*t_corner).m_position.y + 0.1);
    (*t_lamp).m_place = ((*t_corner).m_position - tide_block.tide_u0.xyz);
    {
        var t_tide_0copy2: f32 = (*t_lamp).m_glow;
        tf_Light((*t_lamp).m_place, (*t_corner).m_normal, &t_tide_0copy2);
        (*t_lamp).m_glow = t_tide_0copy2;
    }
    (*t_lamp).m_tint = vec4f(1.0, 0.9, 0.8, 1.0);
    (*t_fog) = tt_Fog((*t_corner).m_position.z, (*t_corner).m_uv);
}

fn ts_Look(t_corner: tt_Corner, t_lamp: tt_FromLamp, t_color: ptr<function, vec4f>) {
    if ((t_lamp.m_glow <= 0.0)) {
        discard;
        return;
    }
    (*t_color) = (t_lamp.m_tint * t_lamp.m_glow);
    (*t_color).a = t_corner.m_uv.x;
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_a2: vec2f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1, tide_a2);
    var tide_handed0: tt_FromLamp = tt_FromLamp(vec3f(), f32(), vec4f());
    var tide_handed1: tt_Fog = tt_Fog(f32(), vec2f());
    ts_Place(&tide_vertex, &tide_handed0, &tide_handed1);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s2 = tide_vertex.m_uv;
    out.tide_h0_1 = tide_handed0.m_glow;
    out.tide_h0_2 = tide_handed0.m_tint;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Look(tt_Corner(vec3f(), vec3f(), in.tide_s2), tt_FromLamp(vec3f(), in.tide_h0_1, in.tide_h0_2), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

    (*t_color) = ((tide_f0 * (1.0 - tf_Haze(t_fog))) * length(t_lamp.m_place));
}

// Fall and Snow on Flake: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_h0_0: f32,
}
struct tt_Flake {
    m_position: vec2f,
}
struct tt_Fog {
    m_depth: f32,
    m_drift: vec2f,
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_saturate1(a: f32) -> f32 {
    if (tide_nan(a)) { return 0.0; }
    return clamp(a, 0.0, 1.0);
}
fn ts_Fall(t_flake: ptr<function, tt_Flake>, t_fog: ptr<function, tt_Fog>) {
    (*t_flake).m_position.y = ((*t_flake).m_position.y - tide_block.tide_u0.x);
    (*t_fog).m_depth = (*t_flake).m_position.y;
    (*t_fog).m_drift = (*t_flake).m_position;
}

fn ts_Snow(t_flake: tt_Flake, t_fog: tt_Fog, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(1.0, 1.0, 1.0, tide_saturate1(t_fog.m_depth));
}

@vertex fn vertex(@location(0) tide_a0: vec2f) -> tide_corner {
    var tide_vertex: tt_Flake = tt_Flake(tide_a0);
    var tide_handed0: tt_Fog = tt_Fog(f32(), vec2f());
    ts_Fall(&tide_vertex, &tide_handed0);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_h0_0 = tide_handed0.m_depth;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Snow(tt_Flake(vec2f()), tt_Fog(in.tide_h0_0, vec2f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);

    }

}
