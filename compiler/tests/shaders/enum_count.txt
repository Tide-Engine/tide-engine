// Lift and Shade on Vertex3: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_a2;
layout(location = 3) in vec4 tide_n0;
layout(location = 4) in vec4 tide_n1;
layout(location = 5) in vec4 tide_n2;
layout(location = 6) in vec4 tide_n3;
struct tt_Vertex3 {
    vec3 m_position;
    vec2 m_uv;
    vec4 m_color;
};
void ts_Game_1Lift(inout tt_Vertex3 t_vertex) {
    t_vertex.m_position.x = (t_vertex.m_position.x + float(4));
}

void main() {
    tt_Vertex3 tide_vertex = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    ts_Game_1Lift(tide_vertex);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
}

// Lift and Shade on Vertex3: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
out vec4 tide_pixel;
struct tt_Vertex3 {
    vec3 m_position;
    vec2 m_uv;
    vec4 m_color;
};
int tide_i(int x) {
    return x;
}
void ts_Game_1Shade(tt_Vertex3 t_vertex, inout vec4 t_color) {
    int t_next = ((int(tide_i(3)) + tide_i(1)) % 4);
    t_color = vec4(float(int(t_next)), float(3), float(4), 1.0);
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Game_1Shade(tt_Vertex3(vec3(0.0), vec2(0.0), vec4(0.0)), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Lift and Shade on Vertex3: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
}
struct tt_Vertex3 {
    m_position: vec3f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_i(x: i32) -> i32 {
    return x;
}
fn ts_Game_1Lift(t_vertex: ptr<function, tt_Vertex3>) {
    (*t_vertex).m_position.x = ((*t_vertex).m_position.x + f32(4));
}

fn ts_Game_1Shade(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    var t_next: i32 = ((i32(tide_i(3)) + tide_i(1)) % 4);
    (*t_color) = vec4f(f32(i32(t_next)), f32(3), f32(4), 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    ts_Game_1Lift(&tide_vertex);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Game_1Shade(tt_Vertex3(vec3f(), vec2f(), vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
