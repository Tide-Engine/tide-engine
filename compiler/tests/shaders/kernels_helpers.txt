@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}

fn ts_Sway(t_corner: ptr<function, tt_Corner>) {

fn tide_get1(tide_i: i32) -> f32 {
    if (tide_i < 0 || tide_i >= tide_count1()) {

fn tide_get0(tide_i: i32) -> tt_Blade {
    if (tide_i < 0 || tide_i >= tide_count0()) {
        return tt_Blade();
    }
    let tide_b = 4u + u32(tide_i) * 4u;

@vertex fn vertex(@location(0) tide_a0: vec3f, @builtin(vertex_index) tide_vertex_index: u32, @builtin(instance_index) tide_instance_index: u32) -> tide_corner {
    let tide_e = 4u + min(tide_instance_index, u32(max(tide_block.tide_v0.y, 1) - 1)) * 4u;
    tide_n0.x = 1.0;
    tide_n0.y = 0.0;
    tide_n0.z = 0.0;
    tide_n0.w = 0.0;
    tide_n1.x = 0.0;
    tide_n1.y = 1.0;
    tide_n1.z = 0.0;
    tide_n1.w = 0.0;
    tide_n2.x = 0.0;
    tide_n2.y = 0.0;
    tide_n2.z = 1.0;
    tide_n2.w = 0.0;
    tide_n3.x = 0.0;
    tide_n3.y = 0.0;
    tide_n3.z = 0.0;
    tide_n3.w = 1.0;
    tide_n4.x = bitcast<f32>(tide_word0(tide_e + 3u));
    let tide_moved = mat4x4f(tide_block.tide_u0, tide_block.tide_u1, tide_block.tide_u2, tide_block.tide_u3) * mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3);
    tide_n0 = tide_moved[0];
    tide_n1 = tide_moved[1];
    tide_n2 = tide_moved[2];
    tide_n3 = tide_moved[3];
    var tide_vertex: tt_Corner = tt_Corner(tide_a0);
    ts_Sway(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_f0 = vec4f(tide_n4.x, 0.0, 0.0, 0.0);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_f0 = in.tide_f0;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Grass(tt_Corner(vec3f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

    return b;

fn ts_CullBlades() {
    var t_blade: tt_Blade = tide_get0(tide_inv_index);
    var t_most: f32 = (tf_Tallest_40() + tf_Tallest_41());
    tf_KeepTall_41(t_blade, t_most);
    tf_Count_42(0);
    tf_Mark_40((tide_inv_index % tide_i(4)));
    tf_Write_43(tide_inv_index, (t_blade.m_height / t_most));
}

@compute @workgroup_size(64, 1, 1) fn compute(@builtin(global_invocation_id) tide_gid: vec3u, @builtin(local_invocation_id) tide_lid: vec3u, @builtin(local_invocation_index) tide_lix: u32, @builtin(workgroup_id) tide_wid: vec3u, @builtin(num_workgroups) tide_groups: vec3u) {
    let tide_size = vec3u(tide_run[4], tide_run[5], tide_run[6]);
    tide_inv_at = vec3i(tide_gid);
    tide_inv_local = vec3i(tide_lid);
    tide_inv_group = vec3i(tide_wid);
    tide_inv_count = vec3i(tide_size);
    tide_inv_index = i32(tide_gid.x + tide_size.x * (tide_gid.y + tide_size.y * tide_gid.z));
    if (all(tide_gid < tide_size)) {
        ts_CullBlades();
    }
    tide_scan[tide_lix] = select(0u, 1u, tide_adding0);
    workgroupBarrier();
    for (var tide_d = 1u; tide_d < 64u; tide_d = tide_d * 2u) {
        var tide_v = 0u;
        if (tide_lix >= tide_d) {
            tide_v = tide_scan[tide_lix - tide_d];
        }
        workgroupBarrier();
        tide_scan[tide_lix] = tide_scan[tide_lix] + tide_v;
        workgroupBarrier();
    }
    let tide_region0 = u32(tide_block.tide_v2.x) + (tide_wid.x + tide_groups.x * (tide_wid.y + tide_groups.y * tide_wid.z)) * 258u;
    if (tide_adding0) {
        let tide_o0 = tide_region0 + 2u + (tide_scan[tide_lix] - 1u) * 4u;
        tide_words[tide_o0 + 0u] = bitcast<u32>(((tide_added0).m_root).x);
        tide_words[tide_o0 + 1u] = bitcast<u32>(((tide_added0).m_root).y);
        tide_words[tide_o0 + 2u] = bitcast<u32>(((tide_added0).m_root).z);
        tide_words[tide_o0 + 3u] = bitcast<u32>((tide_added0).m_height);
    }
    if (tide_lix == 63u) {
        tide_words[tide_region0] = tide_scan[tide_lix];
    }
    workgroupBarrier();
}
