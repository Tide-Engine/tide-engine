    var t_h: mat4x4f = tf_Hip_40(tide_j0.x);
    (*t_corner).m_position = ((t_h * t_m) * vec4f((*t_corner).m_position, tide_f(1.0))).xyz;
    (*t_corner).m_position.y = ((*t_corner).m_position.y + tf_Total_40());
}

fn ts_Lit(t_corner: tt_Boned, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(tf_Total_40(), 1.0, 1.0, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec4i, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f, @location(6) tide_in4: vec4f, @location(7) tide_ij0: vec4i) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_n4 = tide_in4;
    tide_j0 = tide_ij0;
    var tide_vertex: tt_Boned = tt_Boned(tide_a0, tide_a1);
    ts_Play(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Boned(vec3f(), vec4i()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
