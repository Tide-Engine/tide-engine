// Caster on Vertex3: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
}
struct tt_Vertex3 {
    m_position: vec3f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn ts_Caster(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(1.0, 1.0, 1.0, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Caster(tt_Vertex3(vec3f(), vec2f(), vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

// Bent and Lit on Vertex3: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
@group(0) @binding(1) var pixels0: texture_depth_2d_array;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(4) var sampler1: sampler_comparison;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) tide_s0: vec3f,
    @location(1) tide_h0_0: f32,
}
struct tt_Vertex3 {
    m_position: vec3f,
    m_uv: vec2f,
    m_color: vec4f,
}
struct tt_ToFrag {
    m_lit: f32,
}
fn tide_cube_face(d: vec3f) -> vec3f {
    let a = abs(d);
    if (a.x >= a.y && a.x >= a.z) {
        let m = max(a.x, 1e-30);
        if (d.x >= 0.0) { return vec3f((1.0 - d.z / m) * 0.5, (1.0 - d.y / m) * 0.5, 0.0); }
        return vec3f((1.0 + d.z / m) * 0.5, (1.0 - d.y / m) * 0.5, 1.0);
    }
    if (a.y >= a.z) {
        let m = max(a.y, 1e-30);
        if (d.y >= 0.0) { return vec3f((1.0 + d.x / m) * 0.5, (1.0 + d.z / m) * 0.5, 2.0); }
        return vec3f((1.0 + d.x / m) * 0.5, (1.0 - d.z / m) * 0.5, 3.0);
    }
    let m = max(a.z, 1e-30);
    if (d.z >= 0.0) { return vec3f((1.0 + d.x / m) * 0.5, (1.0 - d.y / m) * 0.5, 4.0); }
    return vec3f((1.0 - d.x / m) * 0.5, (1.0 - d.y / m) * 0.5, 5.0);
}
fn tide_depth_cube0_0(d: vec3f) -> f32 {
    let tide_face = tide_cube_face(d);
    return tide_depth(textureSampleLevel(pixels0, sampler0, tide_face.xy, i32(tide_face.z), 0));
}
fn ts_Bent(t_vertex: ptr<function, tt_Vertex3>, t_to: ptr<function, tt_ToFrag>) {
    (*t_to).m_lit = tide_depth_cube0_0(((*t_vertex).m_position - tide_block.tide_u0.xyz));
}

fn tide_compare_cube0_1(d: vec3f, v: f32) -> f32 {
    let tide_face = tide_cube_face(d);
    return textureSampleCompareLevel(pixels0, sampler1, tide_face.xy, i32(tide_face.z), tide_depth(clamp(v, 0.0, 1.0)));
}
fn tf_Shadowed_40(t_way: vec3f, t_depth: f32) -> f32 {
    return tide_compare_cube0_1(t_way, t_depth);
}

fn ts_Lit(t_vertex: tt_Vertex3, t_from: tt_ToFrag, t_color: ptr<function, vec4f>) {
    var t_way: vec3f = (t_vertex.m_position - tide_block.tide_u0.xyz);
    var t_near: f32 = tide_depth_cube0_0(t_way);
    var t_lit: f32 = (tide_compare_cube0_1(t_way, 0.5) + tf_Shadowed_40((-t_way), (t_near - 0.01)));
    (*t_color) = vec4f(t_lit, t_near, t_from.m_lit, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    var tide_handed0: tt_ToFrag = tt_ToFrag(f32());
    ts_Bent(&tide_vertex, &tide_handed0);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s0 = tide_vertex.m_position;
    out.tide_h0_0 = tide_handed0.m_lit;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Vertex3(in.tide_s0, vec2f(), vec4f()), tt_ToFrag(in.tide_h0_0), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
