// Lift and Ground on Vertex3: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_v0: vec4i,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
@group(0) @binding(1) var pixels0: texture_2d_array<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_depth_2d_array;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_2d_array<f32>;
@group(0) @binding(6) var sampler2: sampler;
@group(0) @binding(8) var sampler3: sampler;
@group(0) @binding(10) var sampler4: sampler_comparison;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex3 {
    m_position: vec3f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_Lift(t_vertex: ptr<function, tt_Vertex3>) {
    (*t_vertex).m_position.y = ((*t_vertex).m_position.y + (tide_straight(textureSampleLevel(pixels0, sampler0, (*t_vertex).m_uv, clamp(0, 0, i32(textureNumLayers(pixels0)) - 1), 0.0)).r + textureSampleLevel(pixels1, sampler1, (*t_vertex).m_uv, clamp(3, 0, i32(textureNumLayers(pixels1)) - 1), 0)));
}

fn ts_Ground(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec2f = t_vertex.m_uv;
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);
    var t_layer: i32 = i32((t_vertex.m_uv.x * 4.0));
    var t_said: vec4f = tide_straight(textureSampleLevel(pixels2, sampler2, t_vertex.m_uv, clamp(t_layer, 0, i32(textureNumLayers(pixels2)) - 1), 0.0));
    var t_blended: vec4f = tide_straight(textureSampleLevel(pixels2, sampler0, t_vertex.m_uv, clamp(tide_block.tide_v0.x, 0, i32(textureNumLayers(pixels2)) - 1), 0.0));
    var t_far: vec4f = tide_straight(textureSampleGrad(pixels0, sampler3, t_vertex.m_uv, clamp(2, 0, i32(textureNumLayers(pixels0)) - 1), tide_dx0, tide_dy0));
    var tide_choice0: vec2f = t_vertex.m_uv;
    var tide_choice1: f32 = 0.1;
    var tide_choice2: i32 = 0;
    var tide_choice3: vec2f = vec2f(tide_choice1, 0.0);
    var tide_choice4: i32 = 0;
    var tide_choice5: f32 = 0.1;
    var tide_choice6: vec2f = vec2f(0.0, tide_choice5);
    var t_steep: vec4f = tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice0, clamp(t_layer, 0, i32(textureNumLayers(pixels0)) - 1), tide_choice3, tide_choice6));
    var t_lit: f32 = textureSampleCompareLevel(pixels1, sampler4, t_vertex.m_uv, clamp(tide_block.tide_v0.x, 0, i32(textureNumLayers(pixels1)) - 1), clamp(0.5, 0.0, 1.0));
    var t_depth: f32 = textureSampleLevel(pixels1, sampler1, t_vertex.m_uv, clamp(1, 0, i32(textureNumLayers(pixels1)) - 1), 0);
    (*t_color) = ((t_said * t_blended) + (((t_far * t_steep) * t_lit) * t_depth));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    ts_Lift(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Ground(tt_Vertex3(vec3f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
