// Read on Vertex3: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_2d_array<f32>;

@group(0) @binding(9) var pixels4: texture_2d<f32>;

struct tt_Vertex3 {
    m_position: vec3f,

fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_min1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a < b) { return a; }
    return b;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_clamp1(a: f32, b: f32, c: f32) -> f32 {
    if (tide_nan(a)) { return b; }
    return tide_max1(b, tide_min1(c, a));
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_Levelled_40(t_uv: vec2f) -> vec4f {
    var tide_choice0: vec2f = t_uv;
    var tide_choice1: f32 = 1.5;
    return tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice0, tide_clamp1(tide_choice1, 0.0, 31.0)));
}

fn tf_Graded_40(t_uv: vec2f, t_across: vec2f, t_down: vec2f) -> vec4f {
    var tide_choice2: vec2f = t_uv;
    var tide_choice3: vec2f = t_across;
    var tide_choice4: vec2f = t_down;
    return tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice2, tide_choice3, tide_choice4));
}

fn tf_Layered_40(t_uv: vec2f) -> vec4f {
    return tide_straight(textureSampleLevel(pixels1, sampler1, t_uv, clamp(2, 0, i32(textureNumLayers(pixels1)) - 1), 0.0));
}

fn tf_Filled_40(t_at: vec3f) -> vec4f {
    return tide_straight(textureSampleLevel(pixels2, sampler2, t_at, 0.0));
}

fn tf_Painted_40(t_uv: vec2f) -> vec4f {
    return (tide_straight(textureSampleLevel(pixels3, sampler2, t_uv, 0.0)) + vec4f(tide_straight(textureSampleLevel(pixels4, sampler2, t_uv, 0.0)).r, 0.0, 0.0, 0.0));
}

fn ts_Read(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    let tide_d0: vec2f = dpdx(t_vertex.m_uv);
    let tide_d1: vec2f = dpdy(t_vertex.m_uv);
    (*t_color) = ((((tf_Levelled_40(t_vertex.m_uv) + tf_Graded_40(t_vertex.m_uv, tide_d0, tide_d1)) + tf_Layered_40(t_vertex.m_uv)) + tf_Filled_40(vec3f(t_vertex.m_uv, 0.5))) + tf_Painted_40(t_vertex.m_uv));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Read(tt_Vertex3(vec3f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

struct tt_Coated {
    m_lift: f32,
    m_shade: f32,

fn tide_nan(x: f32) -> bool {

fn tf_Lift_40() -> f32 {
    var tide_choice0: f32 = tide_n4.x;
    var tide_choice1: f32 = 0.5;
    var tide_choice2: vec2f = vec2f(tide_choice1);
    var tide_choice3: i32 = 0;
    var tide_choice4: f32 = tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice2, tide_clamp1(0.0, 0.0, 31.0))).g;
    return (tide_choice0 * tide_choice4);
}

fn tf_Twice(t_x: f32) -> f32 {
    return (t_x * 2.0);
}

fn tf_Kept_40() -> f32 {
    var t_copy: tt_Coated = tt_Coated(tide_n4.x, tide_n4.y);
    return t_copy.m_shade;
}

fn ts_Rise(t_vertex: ptr<function, tt_Vertex3>) {
    (*t_vertex).m_position.y = ((*t_vertex).m_position.y + ((tf_Lift_40() + tf_Twice((*t_vertex).m_position.x)) + tf_Kept_40()));
}

fn tf_Lift_40_8f() -> f32 {
    var tide_choice0: f32 = tide_f0.x;
    var tide_choice1: f32 = 0.5;
    var tide_choice2: vec2f = vec2f(tide_choice1);
    var tide_choice3: i32 = 0;
    var tide_choice4: f32 = tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice2, tide_clamp1(0.0, 0.0, 31.0))).g;
    return (tide_choice0 * tide_choice4);
}

fn tf_Kept_40_8f() -> f32 {
    var t_copy: tt_Coated = tt_Coated(tide_f0.x, tide_f0.y);
    return t_copy.m_shade;
}

fn ts_Coat(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(tf_Lift_40_8f(), tf_Twice(t_vertex.m_uv.x), tf_Kept_40_8f(), 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f, @location(7) tide_in4: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_n4 = tide_in4;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    ts_Rise(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    out.tide_f0 = vec4f(tide_n4.x, tide_n4.y, 0.0, 0.0);
    return out;
}

}

// compute Fill: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }

    tide_none: vec4i,

@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_storage_3d<rgba8unorm, write>;
@group(0) @binding(35) var<storage, read> tide_run: array<u32>;
var<private> tide_inv_index: i32;
var<private> tide_inv_at: vec3i;
var<private> tide_inv_local: vec3i;
var<private> tide_inv_group: vec3i;
var<private> tide_inv_count: vec3i;
fn tide_image_put0(at: vec3i, value: vec4f) {
    if (all(at >= vec3i(0)) && all(at < vec3i(textureDimensions(pixels0)))) {
        textureStore(pixels0, at, vec4f(value.rgb * value.a, value.a));
    }
}
fn tf_Put_40(t_at: vec3i, t_color: vec4f) {
    tide_image_put0(t_at, t_color);
}

fn ts_Fill() {
    tf_Put_40(tide_inv_at, vec4f(vec3f(tide_inv_at), 1.0));
}

@compute @workgroup_size(2, 2, 2) fn compute(@builtin(global_invocation_id) tide_gid: vec3u, @builtin(local_invocation_id) tide_lid: vec3u, @builtin(local_invocation_index) tide_lix: u32, @builtin(workgroup_id) tide_wid: vec3u, @builtin(num_workgroups) tide_groups: vec3u) {
    let tide_size = vec3u(tide_run[4], tide_run[5], tide_run[6]);
    tide_inv_at = vec3i(tide_gid);
    tide_inv_local = vec3i(tide_lid);
    tide_inv_group = vec3i(tide_wid);
    tide_inv_count = vec3i(tide_size);
    tide_inv_index = i32(tide_gid.x + tide_size.x * (tide_gid.y + tide_size.y * tide_gid.z));
    if (all(tide_gid < tide_size)) {
        ts_Fill();
    }
}
