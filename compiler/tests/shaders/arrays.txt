    }
    var tide_all = tide_a;
    return tide_all[tide_i];
}

fn tide_pick0(tide_c: bool, tide_a: tt_Stop, tide_b: tt_Stop) -> tt_Stop {
    if (tide_c) {
        return tide_a;
    }
    return tide_b;
}

fn tf_Total(t_values: array<f32, 4>) -> f32 {
    var t_sum: f32 = 0.0;
    for (var tide_k2: i32 = 0; tide_k2 < 4; tide_k2++) {
        var t_value: f32 = tide_el5(t_values, tide_k2);
        t_sum = (t_sum + t_value);
    }
    return t_sum;
}

fn ts_Rows(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    var tide_array0: array<array<f32, 3>, 2> = array<array<f32, 3>, 2>(array<f32, 3>(tide_block.tide_u4.w, tide_block.tide_u5.x, tide_block.tide_u5.y), array<f32, 3>(tide_block.tide_u5.z, tide_block.tide_u5.w, tide_block.tide_u6.x));
    var tide_array1: array<i32, 2> = array<i32, 2>(tide_block.tide_v0.x, tide_block.tide_v0.y);
    var tide_array2: array<tt_Stop, 3> = array<tt_Stop, 3>(tt_Stop(tide_block.tide_u0, tide_block.tide_u1.x), tt_Stop(vec4f(tide_block.tide_u1.y, tide_block.tide_u1.z, tide_block.tide_u1.w, tide_block.tide_u2.x), tide_block.tide_u2.y), tt_Stop(vec4f(tide_block.tide_u2.z, tide_block.tide_u2.w, tide_block.tide_u3.x, tide_block.tide_u3.y), tide_block.tide_u3.z));
    var tide_array3: array<f32, 4> = array<f32, 4>(tide_block.tide_u3.w, tide_block.tide_u4.x, tide_block.tide_u4.y, tide_block.tide_u4.z);
    var t_mine: array<array<f32, 3>, 2> = tide_array0;
    if ((tide_el0(tide_array1, 1) >= 0 && tide_el0(tide_array1, 1) < 3) && (tide_el0(tide_array1, 0) >= 0 && tide_el0(tide_array1, 0) < 2)) {
        t_mine[tide_el0(tide_array1, 0)][tide_el0(tide_array1, 1)] = 1.0;
    }
    {
    }
    var t_row: array<f32, 3> = tide_el1(t_mine, tide_el0(tide_array1, 0));
    t_mine[0] = t_row;
    var t_made: array<array<i32, 2>, 2> = array<array<i32, 2>, 2>(array<i32, 2>(1, 2), array<i32, 2>(3, 4));
    {
        var t_k: i32 = 0;
        loop {
            if (!((t_k < 4))) {
                break;
            }
            {
                if ((t_k == 1)) {
                    continue;
                }
                t_k = (t_k + 2);
            }
            continuing {
                if (((t_k % tide_i(2)) >= 0 && (t_k % tide_i(2)) < 2)) {
                    t_made[1][(t_k % tide_i(2))] = (tide_el0(tide_el2(t_made, 1), (t_k % tide_i(2))) + 1);
                }
            }
        }
    }
    var t_stop: tt_Stop = tide_pick0((t_corner.m_uv.x > 0.5), tide_el3(tide_array2, 0), tide_el3(tide_array2, 1));
    var t_turn: mat2x2f = tide_pick1((t_corner.m_uv.y > 0.5), mat2x2f(1.0, 0.0, 0.0, 1.0), mat2x2f(0.0, 1.0, 1.0, 0.0));
    var t_picked: array<array<f32, 3>, 2> = tide_pick2((t_corner.m_uv.x > 0.5), t_mine, tide_array0);
    var t_sum: f32 = 0.0;
    for (var tide_k0: i32 = 0; tide_k0 < 2; tide_k0++) {
        var t_each: array<f32, 3> = tide_el1(t_picked, tide_k0);
        for (var tide_k1: i32 = 0; tide_k1 < 3; tide_k1++) {
            var t_value: f32 = tide_el4(t_each, tide_k1);
            t_sum = (t_sum + t_value);
        }
    }
    (*t_color) = ((((t_stop.m_color * t_sum) * (t_turn * t_corner.m_uv).x) * f32(tide_el0(tide_el2(t_made, 1), 0))) * tf_Total(tide_array3));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Rows(tt_Corner(vec3f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
