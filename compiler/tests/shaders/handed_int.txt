// Marked and Read on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) @interpolate(flat) tide_h0_0: i32,
    @location(1) @interpolate(flat) tide_h0_1: vec2i,
    @location(2) @interpolate(flat) tide_h0_2: vec3i,
    @location(3) @interpolate(flat) tide_h0_3: vec4i,
    @location(4) @interpolate(flat) tide_h0_4: i32,
    @location(5) @interpolate(flat) tide_h0_5: i32,
    @location(6) @interpolate(flat) tide_h0_6: i32,
    @location(7) tide_h0_7: f32,
}
struct tt_Corner {
    m_position: vec3f,
}
struct tt_Mark {
    m_which: i32,
    m_pair: vec2i,
    m_triple: vec3i,
    m_four: vec4i,
    m_on: bool,
    m_kind: i32,
    m_small: i32,
    m_blended: f32,
}
fn ts_Marked(t_vertex: ptr<function, tt_Corner>, t_mark: ptr<function, tt_Mark>) {
    (*t_mark).m_which = i32((*t_vertex).m_position.x);
    (*t_mark).m_pair = vec2i((*t_mark).m_which, 2);
    (*t_mark).m_triple = vec3i(1, 2, 3);
    (*t_mark).m_four = vec4i(1, 2, 3, 4);
    (*t_mark).m_on = ((*t_mark).m_which > 0);
    (*t_mark).m_kind = 1;
    (*t_mark).m_small = 255;
    (*t_mark).m_blended = (*t_vertex).m_position.y;
}

fn ts_Read(t_vertex: tt_Corner, t_mark: tt_Mark, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(f32((((t_mark.m_which + t_mark.m_pair.x) + t_mark.m_triple.z) + t_mark.m_four.w)), f32(select(0, 1, t_mark.m_on)), select(0.0, (f32(t_mark.m_small) + t_mark.m_blended), (t_mark.m_kind == 1)), 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_in0: vec4f, @location(2) tide_in1: vec4f, @location(3) tide_in2: vec4f, @location(4) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0);
    var tide_handed0: tt_Mark = tt_Mark(i32(), vec2i(), vec3i(), vec4i(), bool(), i32(), i32(), f32());
    ts_Marked(&tide_vertex, &tide_handed0);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_h0_0 = tide_handed0.m_which;
    out.tide_h0_1 = tide_handed0.m_pair;
    out.tide_h0_2 = tide_handed0.m_triple;
    out.tide_h0_3 = tide_handed0.m_four;
    out.tide_h0_4 = select(0, 1, tide_handed0.m_on);
    out.tide_h0_5 = tide_handed0.m_kind;
    out.tide_h0_6 = tide_handed0.m_small;
    out.tide_h0_7 = tide_handed0.m_blended;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Read(tt_Corner(vec3f()), tt_Mark(in.tide_h0_0, in.tide_h0_1, in.tide_h0_2, in.tide_h0_3, (in.tide_h0_4 != 0), in.tide_h0_5, in.tide_h0_6, in.tide_h0_7), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
