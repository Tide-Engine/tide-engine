// Side on Corner: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
var<private> tide_front: bool;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) @interpolate(flat) tide_mirror: i32,
}
struct tt_Corner {
    m_position: vec3f,
}
fn ts_Side(t_vertex: tt_Corner, t_color: ptr<function, vec4f>) {
    (*t_color) = select(vec4f(0.0, 0.0, 1.0, 1.0), vec4f(1.0, 0.0, 0.0, 1.0), tide_front);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_in0: vec4f, @location(2) tide_in1: vec4f, @location(3) tide_in2: vec4f, @location(4) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    let tide_model = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3);
    let tide_det = tide_model[0].x * (tide_model[1].y * tide_model[2].z - tide_model[1].z * tide_model[2].y) + tide_model[0].y * (tide_model[1].z * tide_model[2].x - tide_model[1].x * tide_model[2].z) + tide_model[0].z * (tide_model[1].x * tide_model[2].y - tide_model[1].y * tide_model[2].x);
    out.tide_mirror = select(0, 1, tide_det < 0.0);
    return out;
}
@fragment fn fragment(in: tide_corner, @builtin(front_facing) tide_facing: bool) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_front = tide_facing != ((in.tide_mirror != 0) != (tide_block.tide_screen.z < 0.0));
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Side(tt_Corner(vec3f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
