fn tide_cube_face(d: vec3f) -> vec3f {
    let a = abs(d);
    if (a.x >= a.y && a.x >= a.z) {
        let m = max(a.x, 1e-30);
        if (d.x >= 0.0) { return vec3f((1.0 - d.z / m) * 0.5, (1.0 - d.y / m) * 0.5, 0.0); }
        return vec3f((1.0 + d.z / m) * 0.5, (1.0 - d.y / m) * 0.5, 1.0);
    }
    if (a.y >= a.z) {
        let m = max(a.y, 1e-30);
        if (d.y >= 0.0) { return vec3f((1.0 + d.x / m) * 0.5, (1.0 + d.z / m) * 0.5, 2.0); }
        return vec3f((1.0 + d.x / m) * 0.5, (1.0 - d.z / m) * 0.5, 3.0);
    }
    let m = max(a.z, 1e-30);
    if (d.z >= 0.0) { return vec3f((1.0 + d.x / m) * 0.5, (1.0 - d.y / m) * 0.5, 4.0); }
    return vec3f((1.0 - d.x / m) * 0.5, (1.0 - d.y / m) * 0.5, 5.0);
}

    var tide_choice6: f32 = (tide_number0_0_13(tide_choice4)).x;
    t_value = (t_value + (tide_choice3 + tide_choice6));
    var t_point: vec4f = (tide_number1_0_13_blend(t_vertex.m_uv) + tide_number1_0_13(t_vertex.m_uv));
    t_value = (t_value + ((tide_number2_0_13(t_vertex.m_uv, 1)).x + (tide_number2_1_13_blend(t_vertex.m_uv, 0)).x));
    var t_position: vec4f = tide_number3_0_13_blend(t_vertex.m_uv, 1);
    var t_material: i32 = (tide_number4_0_5(t_vertex.m_uv, 1) + tf_Material_44(t_vertex.m_uv, 0));
    var t_half: vec2f = ((tide_number5_0_13(t_vertex.m_uv, 1)).xy + (textureSampleLevel(pixels6, sampler0, t_vertex.m_uv, clamp(0, 0, i32(textureNumLayers(pixels6)) - 1), 0.0)).xy);
    t_value = (t_value + ((tide_number7_0_13(vec3f(t_vertex.m_uv, 1.0))).x + tf_Toward_47(vec3f(f32((-1)), t_vertex.m_uv))));
    var t_ray: vec4f = tide_number8_0_13(vec3f(t_vertex.m_uv, f32((-1))));
    var t_id: i32 = tide_number9_0_5(vec3f(t_vertex.m_uv.x, 1.0, t_vertex.m_uv.y));
    var t_fog: vec4f = textureSampleLevel(pixels10, sampler0, vec3f(t_vertex.m_uv, 0.5), 0.0);
    t_value = (t_value + ((textureSampleLevel(pixels11, sampler1, vec3f(t_vertex.m_uv, 0.25), 0.0)).x + tide_number12_0_10(vec3f(t_vertex.m_uv, 0.75))));
    t_value = (t_value + (tide_number13_0_13_blend(vec3f(t_vertex.m_uv, 0.5))).x);
    (*t_color) = vec4f(((((t_value + t_point.x) + t_position.y) + t_half.x) + t_ray.z), f32((t_material + t_id)), t_fog.w, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Read(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

// ReadSource on Vertex: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_index2_0(k: vec2i, n: vec2i) -> vec2i {
    return clamp(k, vec2i(0), n - vec2i(1));
}
fn tide_number0_0_13_blend(uv: vec2f) -> vec4f {
    let tide_uv = clamp(uv, vec2f(0.0), vec2f(1.0));
    let tide_size = vec2i(textureDimensions(pixels0));
    let tide_p = tide_uv * vec2f(tide_size) - 0.5;
    let tide_b = floor(tide_p);
    let tide_t = tide_p - tide_b;
    let tide_i = vec2i(tide_b);
    let tide_c0 = textureLoad(pixels0, tide_index2_0(tide_i + vec2i(0, 0), tide_size), 0);
    let tide_c1 = textureLoad(pixels0, tide_index2_0(tide_i + vec2i(1, 0), tide_size), 0);
    let tide_c2 = textureLoad(pixels0, tide_index2_0(tide_i + vec2i(0, 1), tide_size), 0);
    let tide_c3 = textureLoad(pixels0, tide_index2_0(tide_i + vec2i(1, 1), tide_size), 0);
    let tide_y0 = mix(mix(tide_c0, tide_c1, tide_t.x), mix(tide_c2, tide_c3, tide_t.x), tide_t.y);
    return tide_y0;
}
fn tide_number0_0_10(uv: vec2f) -> f32 {
    let tide_uv = clamp(uv, vec2f(0.0), vec2f(1.0));
    let tide_size = vec2i(textureDimensions(pixels0));
    let tide_at = clamp(vec2i(floor(tide_uv * vec2f(tide_size))), vec2i(0), tide_size - vec2i(1));
    return textureLoad(pixels0, tide_at, 0).x;
}
fn ts_ReadSource(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f((tide_number0_0_13_blend(t_vertex.m_uv)).x, tide_number0_0_10(t_vertex.m_uv), 0.0, 1.0);
}
