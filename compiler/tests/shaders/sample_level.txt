// VertexLevel and Levels on Vertex: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_cube<f32>;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_2d_array<f32>;
@group(0) @binding(6) var sampler2: sampler;
@group(0) @binding(8) var sampler3: sampler;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_min1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a < b) { return a; }
    return b;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_clamp1(a: f32, b: f32, c: f32) -> f32 {
    if (tide_nan(a)) { return b; }
    return tide_max1(b, tide_min1(c, a));
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_VertexLevel(t_vertex: ptr<function, tt_Vertex>) {
    var tide_choice0: vec2f = (*t_vertex).m_uv;
    var tide_choice1: i32 = 1;
    (*t_vertex).m_color = tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice0, tide_clamp1(1.0, 0.0, 31.0)));
}

fn tf_Read_40(t_uv: vec2f, t_level: f32) -> vec4f {
    var tide_choice11: vec2f = t_uv;
    var tide_choice12: f32 = t_level;
    return tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice11, tide_clamp1(tide_choice12, 0.0, 31.0)));
}

fn tf_Forward_40(t_uv: vec2f) -> vec4f {
    return tf_Read_40(t_uv, 1.5);
}

fn ts_Levels(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var tide_choice2: f32 = 1.5;
    var tide_choice3: vec2f = t_vertex.m_uv;
    (*t_color) = tide_straight(textureSampleLevel(pixels0, sampler1, tide_choice3, tide_clamp1(tide_choice2, 0.0, 31.0)));
    var tide_choice4: vec2f = t_vertex.m_uv;
    var tide_choice5: i32 = 1;
    var tide_choice6: vec3f = vec3f(tide_choice4, 1.0);
    var tide_choice7: f32 = 2.5;
    (*t_color) = ((*t_color) + tide_straight(textureSampleLevel(pixels1, sampler2, tide_choice6, tide_clamp1(tide_choice7, 0.0, 31.0))));
    var tide_choice8: vec2f = t_vertex.m_uv;
    var tide_choice9: i32 = 1;
    var tide_choice10: f32 = 2.5;
    (*t_color) = ((*t_color) + tide_straight(textureSampleLevel(pixels2, sampler3, tide_choice8, clamp(tide_choice9, 0, i32(textureNumLayers(pixels2)) - 1), tide_clamp1(tide_choice10, 0.0, 31.0))));
    (*t_color) = ((*t_color) + tf_Forward_40(t_vertex.m_uv));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    ts_VertexLevel(&tide_vertex);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Levels(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

// compute KernelLevel: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
struct tide_uniforms {
    tide_v0: vec4i,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(33) var<storage, read_write> tide_words: array<u32>;
@group(0) @binding(35) var<storage, read> tide_run: array<u32>;
var<private> tide_inv_index: i32;
var<private> tide_inv_at: vec3i;
var<private> tide_inv_local: vec3i;
var<private> tide_inv_group: vec3i;
var<private> tide_inv_count: vec3i;
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_min1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a < b) { return a; }
    return b;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_clamp1(a: f32, b: f32, c: f32) -> f32 {
    if (tide_nan(a)) { return b; }
    return tide_max1(b, tide_min1(c, a));
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tide_base0() -> u32 {
    return u32(tide_block.tide_v0.x);
}

fn tide_word0(tide_k: u32) -> u32 {
    return tide_words[tide_base0() + tide_k];
}

fn tide_count0() -> i32 {
    return i32(min(tide_word0(0u), u32(tide_block.tide_v0.y)));
}

fn tide_capacity0() -> i32 {
    return i32(min(tide_word0(1u), u32(tide_block.tide_v0.y)));
}

fn tide_get0(tide_i: i32) -> vec4f {
    if (tide_i < 0 || tide_i >= tide_count0()) {
        return vec4f();
    }
    let tide_b = 4u + u32(tide_i) * 4u;
    let tide_q0 = vec4i(bitcast<i32>(tide_word0(tide_b + 0u)), bitcast<i32>(tide_word0(tide_b + 1u)), bitcast<i32>(tide_word0(tide_b + 2u)), bitcast<i32>(tide_word0(tide_b + 3u)));
    let tide_p0 = bitcast<vec4f>(tide_q0);
    return tide_p0;
}

fn tide_put0(tide_i: i32, tide_v: vec4f) {
    if (tide_i < 0 || tide_i >= tide_capacity0()) {
        return;
    }
    let tide_b = tide_base0() + 4u + u32(tide_i) * 4u;
    tide_words[tide_b + 0u] = bitcast<u32>((tide_v).x);
    tide_words[tide_b + 1u] = bitcast<u32>((tide_v).y);
    tide_words[tide_b + 2u] = bitcast<u32>((tide_v).z);
    tide_words[tide_b + 3u] = bitcast<u32>((tide_v).w);
}

fn tf_Read_40(t_uv: vec2f, t_level: f32) -> vec4f {
    var tide_choice0: vec2f = t_uv;
    var tide_choice1: f32 = t_level;
    return tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice0, tide_clamp1(tide_choice1, 0.0, 31.0)));
}

fn tf_Forward_40(t_uv: vec2f) -> vec4f {
    return tf_Read_40(t_uv, 1.5);
}

fn ts_KernelLevel() {
    tide_put0(tide_inv_at.x, tf_Forward_40(vec2f(0.5)));
}

@compute @workgroup_size(1, 1, 1) fn compute(@builtin(global_invocation_id) tide_gid: vec3u, @builtin(local_invocation_id) tide_lid: vec3u, @builtin(local_invocation_index) tide_lix: u32, @builtin(workgroup_id) tide_wid: vec3u, @builtin(num_workgroups) tide_groups: vec3u) {
    let tide_size = vec3u(tide_run[4], tide_run[5], tide_run[6]);
    tide_inv_at = vec3i(tide_gid);
    tide_inv_local = vec3i(tide_lid);
    tide_inv_group = vec3i(tide_wid);
    tide_inv_count = vec3i(tide_size);
    tide_inv_index = i32(tide_gid.x + tide_size.x * (tide_gid.y + tide_size.y * tide_gid.z));
    if (all(tide_gid < tide_size)) {
        ts_KernelLevel();
    }
}
