// Read on Vertex: WGSL
const tide_reversed_depth = true;
fn tide_depth(d: f32) -> f32 { return select(d, 1.0 - d, tide_reversed_depth); }
fn tide_clip_depth(p: vec4f) -> f32 { return select((p.z + p.w) * 0.5, (p.w - p.z) * 0.5, tide_reversed_depth); }
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_2d<f32>;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_2d_array<f32>;
@group(0) @binding(6) var sampler2: sampler;
@group(0) @binding(7) var pixels3: texture_2d_array<f32>;
@group(0) @binding(9) var pixels4: texture_cube<f32>;
@group(0) @binding(11) var pixels5: texture_cube<f32>;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_min1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a < b) { return a; }
    return b;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_clamp1(a: f32, b: f32, c: f32) -> f32 {
    if (tide_nan(a)) { return b; }
    return tide_max1(b, tide_min1(c, a));
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_Borrow_41(t_uv: vec2f) -> vec4f {
    var tide_choice4: vec2f = t_uv;
    var tide_choice5: i32 = 0;
    return tide_straight(textureSampleLevel(pixels1, sampler1, tide_choice4, tide_clamp1(0.0, 0.0, 31.0)));
}

fn tf_Forward_41(t_uv: vec2f) -> vec4f {
    return tf_Borrow_41(t_uv);
}

fn ts_Read(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var tide_choice0: vec2f = t_vertex.m_uv;
    var tide_choice1: vec4f = tide_straight(textureSampleLevel(pixels0, sampler0, tide_choice0, 0.0));
    var tide_choice2: vec2f = t_vertex.m_uv;
    var tide_choice3: vec4f = tf_Forward_41(tide_choice2);
    var tide_choice6: vec4f = (tide_choice1 + tide_choice3);
    var tide_choice7: vec2f = t_vertex.m_uv;
    var tide_choice8: f32 = 0.1;
    var tide_choice9: i32 = 0;
    var tide_choice10: vec2f = vec2f(tide_choice8, 0.0);
    var tide_choice11: i32 = 0;
    var tide_choice12: f32 = 0.1;
    var tide_choice13: vec2f = vec2f(0.0, tide_choice12);
    var tide_choice14: vec4f = tide_straight(textureSampleGrad(pixels2, sampler0, tide_choice7, clamp(1, 0, i32(textureNumLayers(pixels2)) - 1), tide_choice10, tide_choice13));
    var tide_choice15: vec4f = (tide_choice6 + tide_choice14);
    var tide_choice16: vec2f = t_vertex.m_uv;
    var tide_choice17: i32 = 0;
    var tide_choice18: i32 = 0;
    var tide_choice19: vec4f = tide_straight(textureSampleLevel(pixels3, sampler0, tide_choice16, clamp(tide_choice17, 0, i32(textureNumLayers(pixels3)) - 1), tide_clamp1(0.0, 0.0, 31.0)));
    var tide_choice20: vec4f = (tide_choice15 + tide_choice19);
    var tide_choice21: vec2f = t_vertex.m_uv;
    var tide_choice22: i32 = 1;
    var tide_choice23: vec3f = vec3f(tide_choice21, 1.0);
    var tide_choice24: i32 = 0;
    var tide_choice25: vec4f = tide_straight(textureSampleLevel(pixels4, sampler2, tide_choice23, tide_clamp1(0.0, 0.0, 31.0)));
    var tide_choice26: vec4f = (tide_choice20 + tide_choice25);
    var tide_choice27: vec2f = t_vertex.m_uv;
    var tide_choice28: i32 = 1;
    var tide_choice29: vec3f = vec3f(tide_choice27, 1.0);
    var tide_choice30: vec4f = tide_straight(textureSampleLevel(pixels5, sampler2, tide_choice29, 0.0));
    (*t_color) = (tide_choice26 + tide_choice30);
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, tide_clip_depth(tide_clip), tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Read(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
