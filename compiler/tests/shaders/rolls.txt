uint tide_xxh_word(uint h, int w) {
    uint k = h + uint(w) * 0xC2B2AE3Du;
    return ((k << 17) | (k >> 15)) * 0x27D4EB2Fu;
}
uint tide_xxh_lane(uint h, int w) {
    uint k = h + uint(w) * 0x85EBCA77u;
    return ((k << 13) | (k >> 19)) * 0x9E3779B1u;

int tide_xxh_end(uint h) {
    uint k = h ^ (h >> 15);
    k = k * 0x85EBCA77u;
    k = k ^ (k >> 13);

    return tide_xxh_end(tide_xxh_word(0x165667B5u, v));

    return tide_xxh_end(tide_xxh_word(tide_xxh_word(tide_xxh_word(0x165667BDu, v.x), v.y), v.z));

uint tide_roll() {
    tide_dice = tide_dice * 747796405u + 2891336453u;
    uint w = ((tide_dice >> ((tide_dice >> 28) + 4u)) ^ tide_dice) * 277803737u;
    return (w >> 22) ^ w;
}
float tide_roll_float() {
    return float(tide_roll() >> 8) * (1.0 / 16777216.0);
}
float tide_roll_float_between(float a, float b) {
    float v = a + (b - a) * tide_roll_float();
    if (v < b) { return v; }
    return a;
}
vec3 tide_roll_float3_direction() {
    float z = tide_roll_float() * 2.0 - 1.0;
    float w = tide_roll_float() * 6.28318531;
    float s = sqrt(max(1.0 - z * z, 0.0));
    return vec3(cos(w) * s, sin(w) * s, z);
}
float tf_Jitter(float t_by) {
    return (tide_roll_float_between((-t_by), t_by) + float((tide_hash1(tide_v0.y) % tide_i(2))));
}

void ts_Shake(inout tt_Corner t_corner) {
    t_corner.m_position = (t_corner.m_position + (tide_roll_float3_direction() * tf_Jitter(0.1)));
}

    out.tide_s1 = tide_vertex.m_uv;

    }
    return tide_color;
