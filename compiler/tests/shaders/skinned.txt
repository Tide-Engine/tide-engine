    return textureLoad(pixels1, vec2i(p & 1023, p >> 10), 0);

fn tide_skin_matrix(i: i32) -> mat4x4f {
    if (i < 0 || i >= tide_block.tide_v0.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let p = (tide_j0.x + i) * 4;
    return mat4x4f(bitcast<vec4f>(tide_skin_words0(p + 0)), bitcast<vec4f>(tide_skin_words0(p + 1)), bitcast<vec4f>(tide_skin_words0(p + 2)), bitcast<vec4f>(tide_skin_words0(p + 3)));
}
fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}

    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_j0 = tide_ij0;

    let tide_skin = tide_skinning(i32(tide_vertex_index));

    out.tide_s2 = tide_vertex.m_color;
    return out;
}

fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tide_normals(m: mat4x4f) -> mat3x3f {
    var a: vec3f = cross(m[1].xyz, m[2].xyz);
    var b: vec3f = cross(m[2].xyz, m[0].xyz);
    var c: vec3f = cross(m[0].xyz, m[1].xyz);
    if (dot(m[0].xyz, a) < 0.0) {
        return mat3x3f(-a, -b, -c);
    }
    return mat3x3f(a, b, c);
}
fn ts_Deform(t_corner: ptr<function, tt_Corner>) {
    (*t_corner).m_position = ((*t_corner).m_position + ((*t_corner).m_normal * (*t_corner).m_glow));
    (*t_corner).m_color = vec4f((((*t_corner).m_tangent.xyz * 0.5) + vec3f(0.5)), 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_a2: vec4f, @location(3) tide_a3: vec4f, @location(4) tide_a4: f32, @location(5) tide_in0: vec4f, @location(6) tide_in1: vec4f, @location(7) tide_in2: vec4f, @location(8) tide_in3: vec4f, @location(9) tide_ij0: vec4i, @builtin(vertex_index) tide_vertex_index: u32, @builtin(instance_index) tide_instance_index: u32) -> tide_corner {

    tide_vertex.m_normal = tide_normalize3(tide_normals(tide_skin) * tide_vertex.m_normal);
    var tide_tangent = (tide_skin * vec4f(tide_vertex.m_tangent.xyz, 0.0)).xyz;
    tide_tangent = tide_tangent - tide_vertex.m_normal * dot(tide_vertex.m_normal, tide_tangent);
    let tide_side = select(1.0, -1.0, dot(tide_skin[0].xyz, cross(tide_skin[1].xyz, tide_skin[2].xyz)) < 0.0);
    tide_vertex.m_tangent = vec4f(tide_normalize3(tide_tangent), tide_vertex.m_tangent.w * tide_side);
    ts_Deform(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_normal;
    out.tide_s3 = tide_vertex.m_color;
    let tide_turned = tide_normals(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3));
    out.tide_normals0 = tide_turned[0];
    out.tide_normals1 = tide_turned[1];
    out.tide_normals2 = tide_turned[2];
    return out;
}

struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
    m_tangent: vec4f,
    m_color: vec4f,
    m_glow: f32,
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tide_normals(m: mat4x4f) -> mat3x3f {
    var a: vec3f = cross(m[1].xyz, m[2].xyz);
    var b: vec3f = cross(m[2].xyz, m[0].xyz);
    var c: vec3f = cross(m[0].xyz, m[1].xyz);
    if (dot(m[0].xyz, a) < 0.0) {
        return mat3x3f(-a, -b, -c);
    }
    return mat3x3f(a, b, c);
}
fn ts_Deform(t_corner: ptr<function, tt_Corner>) {
    (*t_corner).m_position = ((*t_corner).m_position + ((*t_corner).m_normal * (*t_corner).m_glow));
    (*t_corner).m_color = vec4f((((*t_corner).m_tangent.xyz * 0.5) + vec3f(0.5)), 1.0);
}

fn tide_skin_words0(p: i32) -> vec4i {
    return textureLoad(pixels0, vec2i(p & 1023, p >> 10), 0);
}
fn tide_skin_words1(p: i32) -> vec4i {
    return textureLoad(pixels1, vec2i(p & 1023, p >> 10), 0);
}
fn tide_skin_matrix(i: i32) -> mat4x4f {
    if (i < 0 || i >= tide_block.tide_v0.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let p = (tide_j0.x + i) * 4;
    return mat4x4f(bitcast<vec4f>(tide_skin_words0(p + 0)), bitcast<vec4f>(tide_skin_words0(p + 1)), bitcast<vec4f>(tide_skin_words0(p + 2)), bitcast<vec4f>(tide_skin_words0(p + 3)));
}
fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}
@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_a2: vec4f, @location(3) tide_a3: vec4f, @location(4) tide_a4: f32, @location(5) tide_in0: vec4f, @location(6) tide_in1: vec4f, @location(7) tide_in2: vec4f, @location(8) tide_in3: vec4f, @location(9) tide_ij0: vec4i, @builtin(vertex_index) tide_vertex_index: u32, @builtin(instance_index) tide_instance_index: u32) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_j0 = tide_ij0;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1, tide_a2, tide_a3, tide_a4);
    let tide_skin = tide_skinning(i32(tide_vertex_index));
    tide_vertex.m_position = (tide_skin * vec4f(tide_vertex.m_position, 1.0)).xyz;
    tide_vertex.m_normal = tide_normalize3(tide_normals(tide_skin) * tide_vertex.m_normal);
    var tide_tangent = (tide_skin * vec4f(tide_vertex.m_tangent.xyz, 0.0)).xyz;
    tide_tangent = tide_tangent - tide_vertex.m_normal * dot(tide_vertex.m_normal, tide_tangent);
    let tide_side = select(1.0, -1.0, dot(tide_skin[0].xyz, cross(tide_skin[1].xyz, tide_skin[2].xyz)) < 0.0);
    tide_vertex.m_tangent = vec4f(tide_normalize3(tide_tangent), tide_vertex.m_tangent.w * tide_side);
    ts_Deform(&tide_vertex);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s3 = tide_vertex.m_color;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    tide_color = in.tide_s3;
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}

    var l: f32 = dot(v, v);

fn tide_normals(m: mat4x4f) -> mat3x3f {
    var a: vec3f = cross(m[1].xyz, m[2].xyz);
    var b: vec3f = cross(m[2].xyz, m[0].xyz);
    var c: vec3f = cross(m[0].xyz, m[1].xyz);

    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}
fn ts_Tint(t_corner: tt_Vertex3, t_color: ptr<function, vec4f>) {
    (*t_color) = (t_corner.m_color * tide_f0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f, @location(7) tide_in4: vec4f, @location(8) tide_ij0: vec4i, @builtin(vertex_index) tide_vertex_index: u32, @builtin(instance_index) tide_instance_index: u32) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_n4 = tide_in4;
    tide_j0 = tide_ij0;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    let tide_skin = tide_skinning(i32(tide_vertex_index));
    tide_vertex.m_position = (tide_skin * vec4f(tide_vertex.m_position, 1.0)).xyz;
    var tide_handed0: tt_PlacedInfo = tt_PlacedInfo(f32());
    var tide_moved: vec4f = vec4f();
    ts_Explicit(&tide_vertex, &tide_handed0, &tide_moved);
    let tide_clip = tide_moved;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s2 = tide_vertex.m_color;
    out.tide_f0 = vec4f(tide_n4.x, tide_n4.y, tide_n4.z, tide_n4.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_f0 = in.tide_f0;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Tint(tt_Vertex3(vec3f(), vec2f(), in.tide_s2), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_v0: vec4i,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}

fn tide_skin_matrix(i: i32) -> mat4x4f {
    if (i < 0 || i >= tide_block.tide_v0.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let p = (tide_j0.x + i) * 4;
    return mat4x4f(bitcast<vec4f>(tide_skin_words0(p + 0)), bitcast<vec4f>(tide_skin_words0(p + 1)), bitcast<vec4f>(tide_skin_words0(p + 2)), bitcast<vec4f>(tide_skin_words0(p + 3)));
}
fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }

    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}
