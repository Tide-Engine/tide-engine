layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    ivec4 tide_v0;
};
uniform highp isampler2D pixels0;
uniform highp isampler2D pixels1;
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_a2;
layout(location = 3) in vec4 tide_n0;
layout(location = 4) in vec4 tide_n1;
layout(location = 5) in vec4 tide_n2;
layout(location = 6) in vec4 tide_n3;
layout(location = 7) in ivec4 tide_j0;
out vec4 tide_s2;
struct tt_Vertex3 {
    vec3 m_position;
    vec2 m_uv;
    vec4 m_color;
};
ivec4 tide_skin_words0(int p) {
    return texelFetch(pixels0, ivec2(p & 1023, p >> 10), 0);
}
ivec4 tide_skin_words1(int p) {
    return texelFetch(pixels1, ivec2(p & 1023, p >> 10), 0);
}
mat4 tide_skin_matrix(int i) {
    if (i < 0 || i >= tide_v0.x) return mat4(1.0);
    int p = (tide_j0.x + i) * 4;
    return mat4(intBitsToFloat(tide_skin_words0(p + 0)), intBitsToFloat(tide_skin_words0(p + 1)), intBitsToFloat(tide_skin_words0(p + 2)), intBitsToFloat(tide_skin_words0(p + 3)));
}
mat4 tide_skinning(int vertex) {
    if (vertex < 0 || vertex >= tide_v0.y) return mat4(1.0);
    ivec2 range = tide_skin_words1(vertex).xy;
    int entries = tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) return mat4(1.0);
    vec4 c0 = vec4(0.0);
    vec4 c1 = vec4(0.0);
    vec4 c2 = vec4(0.0);
    vec4 c3 = vec4(0.0);
    float total = 0.0;
    for (int n = 0; n < range.y; n++) {
        int i = range.x + n;
        ivec4 pair = tide_skin_words1(tide_v0.y + (i >> 1));
        int joint = (i & 1) == 0 ? pair.x : pair.z;
        float weight = intBitsToFloat((i & 1) == 0 ? pair.y : pair.w);
        mat4 matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) return mat4(1.0);
    return mat4(c0 / total, c1 / total, c2 / total, c3 / total);
}
void main() {
    tt_Vertex3 tide_vertex = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    mat4 tide_skin = tide_skinning(gl_VertexID);
    tide_vertex.m_position = (tide_skin * vec4(tide_vertex.m_position, 1.0)).xyz;
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s2 = tide_vertex.m_color;
}

fn tide_skin_matrix(i: i32) -> mat4x4f {
    if (i < 0 || i >= tide_block.tide_v0.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let p = (tide_j0.x + i) * 4;
    return mat4x4f(bitcast<vec4f>(tide_skin_words0(p + 0)), bitcast<vec4f>(tide_skin_words0(p + 1)), bitcast<vec4f>(tide_skin_words0(p + 2)), bitcast<vec4f>(tide_skin_words0(p + 3)));

fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}
@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f, @location(7) tide_ij0: vec4i, @builtin(vertex_index) tide_vertex_index: u32, @builtin(instance_index) tide_instance_index: u32) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    tide_j0 = tide_ij0;

    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);

    out.tide_s2 = tide_vertex.m_color;

}

struct tt_Corner {

    float m_glow;

vec3 tide_normalize3(vec3 v) {
    float l = dot(v, v);
    if (l > 1.17549435e-38) { return v * inversesqrt(l); }
    return vec3(0.0);
}
mat3 tide_normals(mat4 m) {
    vec3 a = cross(m[1].xyz, m[2].xyz);
    vec3 b = cross(m[2].xyz, m[0].xyz);
    vec3 c = cross(m[0].xyz, m[1].xyz);
    if (dot(m[0].xyz, a) < 0.0) {
        return mat3(-a, -b, -c);
    }
    return mat3(a, b, c);
}
void ts_Deform(inout tt_Corner t_corner) {
    t_corner.m_position = (t_corner.m_position + (t_corner.m_normal * t_corner.m_glow));
    t_corner.m_color = vec4(((t_corner.m_tangent.xyz * 0.5) + vec3(0.5)), 1.0);
}

fn tide_skin_matrix(i: i32) -> mat4x4f {

fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
