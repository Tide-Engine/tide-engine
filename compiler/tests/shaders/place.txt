// Snap on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
}
fn ts_Snap(t_corner: ptr<function, tt_Corner>, t_clip: ptr<function, vec4f>) {
    var t_at: vec4f = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f((*t_corner).m_position, 1.0));
    var t_half: vec2f = (tide_block.tide_u0.xy * 0.5);
    (*t_clip) = vec4f(((round(((t_at.xy / t_at.w) * t_half)) / t_half) * t_at.w), t_at.z, t_at.w);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    var tide_moved: vec4f = vec4f();
    ts_Snap(&tide_vertex, &tide_moved);
    let tide_clip = tide_moved;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

// Far and Look on Corner: WGSL
struct tide_uniforms {

    return tide_block.tide_camera * (relative * vertex);

var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_h0_0: vec3f,
    @location(1) tide_world: vec3f,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
}
struct tt_FromLamp {
    m_place: vec3f,
}
fn ts_Far(t_corner: ptr<function, tt_Corner>, t_clip: ptr<function, vec4f>, t_lamp: ptr<function, tt_FromLamp>) {
    (*t_corner).m_position.y = ((*t_corner).m_position.y + 1.0);
    (*t_lamp).m_place = (*t_corner).m_position;
    (*t_clip) = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f((*t_corner).m_position, 1.0));
    (*t_clip).z = (*t_clip).w;
}

fn ts_Look(t_corner: tt_Corner, t_lamp: tt_FromLamp, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(t_lamp.m_place.x, tide_world.y, 0.0, 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    var tide_handed0: tt_FromLamp = tt_FromLamp(vec3f());
    var tide_moved: vec4f = vec4f();
    ts_Far(&tide_vertex, &tide_moved, &tide_handed0);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_moved;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_h0_0 = tide_handed0.m_place;
    out.tide_world = tide_place.xyz;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_world = in.tide_world;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Look(tt_Corner(vec3f(), vec3f()), tt_FromLamp(in.tide_h0_0), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);

        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
