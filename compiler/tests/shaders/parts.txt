    let tide_r = textureLoad(pixels0, vec2i((tide_first + 0) & 1023, (tide_first + 0) >> 10u), 0);
    let tide_q0 = vec4i(tide_r[tide_word & 3], 0, 0, 0);
    let tide_p0 = bitcast<vec4f>(tide_q0);
    return tt_Gear(vec4f(), vec3f(), vec3f(), tt_Notch(tide_p0.x, i32()), i32(), array<f32, 2>(), bool());
}

fn tf_Deepest_40() -> f32 {
    var t_deep: f32 = 0.0;
    for (var tide_k2: i32 = 0; tide_k2 < tide_block.tide_v0.x; tide_k2++) {
        var t_gear: tt_Gear = tide_list0(tide_k2);
        t_deep = tide_max1(t_deep, t_gear.m_notch.m_deep);
    }
    return t_deep;
}

fn tide_find2(tide_key: i32) -> i32 {
    let tide_slots = tide_block.tide_v0.w;
    if (tide_slots <= 0) {
        return -1;
    }
    let tide_base = ((tide_block.tide_v0.z * 8 + 3) >> 2u) << 2u;
    var tide_s = tide_hash1(tide_key) & (tide_slots - 1);
    for (var tide_n: i32 = 0; tide_n < tide_slots; tide_n++) {
        let tide_w = tide_base + tide_s;
        let tide_slot = textureLoad(pixels2, vec2i((tide_w >> 2u) & 1023, tide_w >> 12u), 0);
        let tide_e = tide_slot[tide_w & 3];
        if (tide_e == 0) {
            return -1;
        }
        let tide_k = (tide_e - 1) * 8 + 3;
        let tide_held = textureLoad(pixels2, vec2i((tide_k >> 2u) & 1023, tide_k >> 12u), 0);
        if (tide_held[tide_k & 3] == tide_key) {
            return tide_e - 1;
        }
        tide_s = (tide_s + 1) & (tide_slots - 1);
    }
    return -1;
}

fn ts_Rest(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    var t_kinds: i32 = 0;
    for (var tide_k0: i32 = 0; tide_k0 < tide_block.tide_v0.z; tide_k0++) {
        var t_entry: tt_Entry_5int_7Gear_6 = tide_list2(tide_k0);
        t_kinds = (t_kinds + (t_entry.m_value.m_kind + t_entry.m_key));
    }
    var t_seen: i32 = 0;
    for (var tide_k1: i32 = 0; tide_k1 < tide_block.tide_v0.y; tide_k1++) {
        var t_gear: tt_Gear = tide_list1(tide_k1);
        t_seen = (t_seen + 1);
    }
    var t_cell: tt_Gear = tide_list3(vec2i(2, 1));
    (*t_color) = vec4f((tf_Deepest_40() + tide_list2(tide_find2(7)).m_value.m_at.z), ((f32(t_kinds) / tide_f(16.0)) + t_cell.m_at.x), ((f32(t_seen) / tide_f(8.0)) + f32(tide_block.tide_v0.y)), 1.0);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_in0: vec4f, @location(2) tide_in1: vec4f, @location(3) tide_in2: vec4f, @location(4) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Rest(tt_Corner(vec3f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
