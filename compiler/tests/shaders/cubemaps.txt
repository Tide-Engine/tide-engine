// Shine on Corner: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform highp samplerCube pixels0;
uniform highp samplerCube pixels1;
uniform highp samplerCube pixels2;
uniform sampler2D pixels3;
uniform highp samplerCube pixels4;
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec3 tide_a1;
layout(location = 2) in vec4 tide_n0;
layout(location = 3) in vec4 tide_n1;
layout(location = 4) in vec4 tide_n2;
layout(location = 5) in vec4 tide_n3;
out vec3 tide_s1;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
};
void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_normal;
}

// Shine on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform highp samplerCube pixels0;
uniform highp samplerCube pixels1;
uniform highp samplerCube pixels2;
uniform sampler2D pixels3;
uniform highp samplerCube pixels4;
in vec3 tide_s1;
out vec4 tide_pixel;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
};
vec3 tide_normalize3(vec3 v) {
    float l = dot(v, v);
    if (l > 1.17549435e-38) { return v * inversesqrt(l); }
    return vec3(0.0);
}
vec4 tide_straight(vec4 c) {
    if (c.a == 0.0) { return vec4(0.0); }
    return vec4(c.rgb / c.a, c.a);
}
void ts_Shine(tt_Corner t_corner, inout vec4 t_color) {
    vec3 tide_uv0 = (tide_normalize3(t_corner.m_normal));
    vec3 tide_dx0 = dFdx(tide_uv0);
    vec3 tide_dy0 = dFdy(tide_uv0);
    vec3 t_away = tide_normalize3(t_corner.m_normal);
    vec4 t_near = tide_straight(textureLod(pixels0, t_away, 0.0));
    if ((t_near.a < 1.0)) {
        t_near = tide_straight(textureLod(pixels1, (-t_away), 0.0));
    }
    vec4 t_far = vec4(0.0, 0.0, 0.0, 1.0);
    if ((t_corner.m_normal.y > 0.0)) {
        t_far = tide_straight(textureGrad(pixels2, t_away, tide_dx0, tide_dy0));
    }
    t_color = (((tide_straight(textureLod(pixels3, t_corner.m_normal.xy, 0.0)) * t_near) + t_far) + tide_straight(textureLod(pixels4, vec3(0.0, 1.0, 0.0), 0.0)));
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Shine(tt_Corner(vec3(0.0), tide_s1), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Shine on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_cube<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_cube<f32>;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_cube<f32>;
@group(0) @binding(6) var sampler2: sampler;
@group(0) @binding(7) var pixels3: texture_2d<f32>;
@group(0) @binding(8) var sampler3: sampler;
@group(0) @binding(9) var pixels4: texture_cube<f32>;
@group(0) @binding(10) var sampler4: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec3f,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_Shine(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec3f = (tide_normalize3(t_corner.m_normal));
    let tide_dx0: vec3f = dpdx(tide_uv0);
    let tide_dy0: vec3f = dpdy(tide_uv0);
    var t_away: vec3f = tide_normalize3(t_corner.m_normal);
    var t_near: vec4f = tide_straight(textureSampleLevel(pixels0, sampler0, t_away, 0.0));
    if ((t_near.a < 1.0)) {
        t_near = tide_straight(textureSampleLevel(pixels1, sampler1, (-t_away), 0.0));
    }
    var t_far: vec4f = vec4f(0.0, 0.0, 0.0, 1.0);
    if ((t_corner.m_normal.y > 0.0)) {
        t_far = tide_straight(textureSampleGrad(pixels2, sampler2, t_away, tide_dx0, tide_dy0));
    }
    (*t_color) = (((tide_straight(textureSampleLevel(pixels3, sampler3, t_corner.m_normal.xy, 0.0)) * t_near) + t_far) + tide_straight(textureSampleLevel(pixels4, sampler4, vec3f(0.0, 1.0, 0.0), 0.0)));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_normal;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Shine(tt_Corner(vec3f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
