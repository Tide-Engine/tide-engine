fn tf_SamplePair_40(t_uv: vec2f) -> vec4f {
    return (textureSampleLevel(pixels0, sampler0, t_uv, 0.0) + textureSampleLevel(pixels1, sampler1, t_uv, 0.0));
}

fn tf_Forward_3_0Texture_0float2_40(t_uv: vec2f) -> vec4f {
    return tf_SamplePair_40(t_uv);
}

fn tf_SamplePair_42(t_uv: vec2f) -> vec4f {
    return (textureSampleLevel(pixels2, sampler2, t_uv, 0.0) + textureSampleLevel(pixels3, sampler1, t_uv, 0.0));
}

fn tf_Forward_3_0Texture_0float2_42(t_uv: vec2f) -> vec4f {
    return tf_SamplePair_42(t_uv);

fn tf_Moved(t_uv: vec2f) -> vec2f {
    return ((t_uv * 0.5) + vec2f(0.25, 0.25));
}

fn tf_SamplePair_45(t_uv: vec2f) -> vec4f {
    return (textureSampleLevel(pixels5, sampler5, t_uv, 0.0) + textureSampleLevel(pixels6, sampler1, t_uv, 0.0));
}

fn tf_Forward_3_0Texture_0float2_45(t_uv: vec2f) -> vec4f {
    return tf_SamplePair_45(t_uv);
}

fn ts_ReadFields(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec2f = tf_Moved(t_vertex.m_uv);
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);
    (*t_color) = ((((tf_Forward_3_0Texture_0float2_40(t_vertex.m_uv) + tf_Forward_3_0Texture_0float2_42(t_vertex.m_uv)) + textureSampleGrad(pixels4, sampler4, tf_Moved(t_vertex.m_uv), tide_dx0, tide_dy0)) + tf_Forward_3_0Texture_0float2_45(t_vertex.m_uv)) + vec4f(f32(tide_block.tide_v0.xy.x), 0.0, 0.0, 1.0));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_ReadFields(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
