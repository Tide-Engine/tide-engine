layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_n0;

layout(location = 5) in vec4 tide_n3;
out vec3 tide_world;
struct tt_Corner {
    vec3 m_position;
    vec2 m_uv;
};
void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1);

    gl_Position = tide_camera * tide_place;
    tide_world = tide_place.xyz;
}

// Cutout on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
in vec3 tide_world;
out vec4 tide_pixel;
struct tt_Corner {
    vec3 m_position;
    vec2 m_uv;
};
void ts_Cutout(tt_Corner t_vertex, inout vec4 t_color) {
    if (((vec2(gl_FragCoord.x, tide_screen.z - gl_FragCoord.y) / tide_screen.xy).x < 4.0)) {
        discard;
    } else {
        t_color = vec4(tide_world.x, tide_world.y, gl_FragCoord.z, 1.0);
    }
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Cutout(tt_Corner(vec3(0.0), vec2(0.0)), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

    @builtin(position) tide_at: vec4f,

    m_position: vec3f,
    m_uv: vec2f,

        discard;
        return;

        (*t_color) = vec4f(tide_world.x, tide_world.y, tide_at.z, 1.0);
    }
}

}

void ts_Tiled(tt_Corner t_vertex, inout vec4 t_color) {
