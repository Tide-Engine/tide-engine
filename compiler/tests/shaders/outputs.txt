float tide_f(float x) {
    return x;
}
bool tide_nan(float x) {
    return (floatBitsToInt(x) & 0x7fffffff) > 0x7f800000;
}
float tide_max1(float a, float b) {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
void ts_Light(tt_Vertex t_vertex, inout vec4 t_color) {
    vec4 t_normal = textureLod(pixels0, t_vertex.m_uv, 0.0);
    t_color = (textureLod(pixels1, t_vertex.m_uv, 0.0) * tide_max1(t_normal.b, tide_f(0.1)));
}

// Light on Vertex: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_2d<f32>;
@group(0) @binding(4) var sampler1: sampler;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_max1(a: f32, b: f32) -> f32 {
    if (tide_nan(b)) { return a; }
    if (tide_nan(a)) { return b; }
    if (a > b) { return a; }
    return b;
}
fn ts_Light(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var t_normal: vec4f = textureSampleLevel(pixels0, sampler0, t_vertex.m_uv, 0.0);
    (*t_color) = (textureSampleLevel(pixels1, sampler1, t_vertex.m_uv, 0.0) * tide_max1(t_normal.b, tide_f(0.1)));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Light(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb * tide_color.a + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
