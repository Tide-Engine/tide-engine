// Shade on Vertex: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
layout(location = 0) in vec2 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_a2;
struct tt_Vertex {
    vec2 m_position;
    vec2 m_uv;
    vec4 m_color;
};
void main() {
    tt_Vertex tide_vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    vec4 tide_place = vec4(tide_vertex.m_position, 0.0, 1.0);
    gl_Position = tide_camera * tide_place;
}

// Shade on Vertex: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
out vec4 tide_pixel;
struct tt_Vertex {
    vec2 m_position;
    vec2 m_uv;
    vec4 m_color;
};
float tf_Climb(inout int t_at) {
    int tide_step2 = t_at;
    t_at = (t_at + 1);
    return (float(tide_step2) * 0.25);
}

void ts_Shade(tt_Vertex t_vertex, inout vec4 t_color) {
    int t_i = 1;
    float t_level = 0.5;
    t_i = (t_i + 1);
    int tide_step0 = t_i;
    float t_shade = (float(tide_step0) * 0.5);
    float tide_step1 = t_level;
    t_level = (t_level - 1.0);
    float t_was = tide_step1;
    t_color = vec4(t_shade, t_was, tf_Climb(t_i), 1.0);
    int tide_step3 = t_i;
    t_i = (t_i + 1);
    t_color.a = ((tide_step3 > 2) ? 1.0 : 0.5);
    t_i = (t_i - 1);
    int tide_step4 = t_i;
    if ((tide_step4 > 1)) {
        t_color.r = t_level;
    }
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Shade(tt_Vertex(vec2(0.0), vec2(0.0), vec4(0.0)), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Shade on Vertex: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tf_Climb(t_at: ptr<function, i32>) -> f32 {
    var tide_step2: i32 = (*t_at);
    (*t_at) = ((*t_at) + 1);
    return (f32(tide_step2) * 0.25);
}

fn ts_Shade(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var t_i: i32 = 1;
    var t_level: f32 = 0.5;
    t_i = (t_i + 1);
    var tide_step0: i32 = t_i;
    var t_shade: f32 = (f32(tide_step0) * 0.5);
    var tide_step1: f32 = t_level;
    t_level = (t_level - 1.0);
    var t_was: f32 = tide_step1;
    (*t_color) = vec4f(t_shade, t_was, tf_Climb(&t_i), 1.0);
    var tide_step3: i32 = t_i;
    t_i = (t_i + 1);
    (*t_color).a = select(0.5, 1.0, (tide_step3 > 2));
    t_i = (t_i - 1);
    var tide_step4: i32 = t_i;
    if ((tide_step4 > 1)) {
        (*t_color).r = t_level;
    }
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Shade(tt_Vertex(vec2f(), vec2f(), vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
