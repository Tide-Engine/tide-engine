fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_CapacityReserve(t_vertex: ptr<function, tt_Vertex>) {
    var t_low: f32 = (((((((tide_straight(textureSampleLevel(pixels0, sampler0, (*t_vertex).m_uv, 0.0)).r + tide_straight(textureSampleLevel(pixels1, sampler1, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels2, sampler2, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels3, sampler3, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels4, sampler4, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels5, sampler5, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels6, sampler6, (*t_vertex).m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels7, sampler7, (*t_vertex).m_uv, 0.0)).r);
    if ((t_low < 7.5)) {
        (*t_vertex).m_position = ((*t_vertex).m_position + vec2f(1000.0));
    }
}

fn tide_list8(tide_i: i32) -> i32 {
    if (tide_i < 0 || tide_i >= tide_block.tide_v0.x) {
        return i32();
    }
    let tide_word = tide_i * 1;
    let tide_first = tide_word >> 2u;
    let tide_r = textureLoad(pixels8, vec2i((tide_first + 0) & 1023, (tide_first + 0) >> 10u), 0);
    let tide_q0 = vec4i(tide_r[tide_word & 3], 0, 0, 0);
    return tide_q0.x;
}

fn ts_CapacityKinds(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var t_number: f32 = f32(tide_list8(0));
    var t_depth: f32 = textureSampleLevel(pixels9, sampler9, t_vertex.m_uv, 0);
    var t_compared: f32 = textureSampleCompareLevel(pixels10, sampler10, t_vertex.m_uv, clamp(0.5, 0.0, 1.0));
    var t_cube: vec4f = tide_straight(textureSampleLevel(pixels11, sampler11, vec3f(1.0, 0.0, 0.0), 0.0));
    (*t_color) = select(vec4f(1.0, 0.0, 0.0, 1.0), vec4f(0.0, 1.0, 0.0, 1.0), (((((t_number == 7.0) && (t_depth > 0.99)) && (t_compared > 0.99)) && (t_cube.r > 0.99)) && (t_cube.g < 0.01)));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    ts_CapacityReserve(&tide_vertex);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_CapacityKinds(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}

fn tide_skin_matrix(i: i32) -> mat4x4f {
    if (i < 0 || i >= tide_block.tide_v0.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let p = (tide_j0.x + i) * 4;
    return mat4x4f(bitcast<vec4f>(tide_skin_words0(p + 0)), bitcast<vec4f>(tide_skin_words0(p + 1)), bitcast<vec4f>(tide_skin_words0(p + 2)), bitcast<vec4f>(tide_skin_words0(p + 3)));
}
fn tide_skinning(vertex: i32) -> mat4x4f {
    if (vertex < 0 || vertex >= tide_block.tide_v0.y) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    let range = tide_skin_words1(vertex).xy;
    let entries = tide_block.tide_v0.z;
    if (range.x < 0 || range.y < 0 || range.x > entries || range.y > entries - range.x) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    var c0 = vec4f(0.0);
    var c1 = vec4f(0.0);
    var c2 = vec4f(0.0);
    var c3 = vec4f(0.0);
    var total = 0.0;
    for (var n = 0; n < range.y; n++) {
        let i = range.x + n;
        let pair = tide_skin_words1(tide_block.tide_v0.y + (i >> 1));
        let joint = select(pair.x, pair.z, (i & 1) != 0);
        let weight = bitcast<f32>(select(pair.y, pair.w, (i & 1) != 0));
        let matrix = tide_skin_matrix(joint);
        c0 = c0 + matrix[0] * weight;
        c1 = c1 + matrix[1] * weight;
        c2 = c2 + matrix[2] * weight;
        c3 = c3 + matrix[3] * weight;
        total = total + weight;
    }
    if (!(total > 0.0)) { return mat4x4f(vec4f(1,0,0,0),vec4f(0,1,0,0),vec4f(0,0,1,0),vec4f(0,0,0,1)); }
    return mat4x4f(c0 / total, c1 / total, c2 / total, c3 / total);
}

    let tide_dx3: vec2f = dpdx(tide_uv3);

    var t_low: f32 = ((((((((tide_straight(textureSampleLevel(pixels0, sampler0, t_vertex.m_uv, 0.0)).r + tide_straight(textureSampleLevel(pixels1, sampler1, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels2, sampler2, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels3, sampler3, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels4, sampler4, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels5, sampler5, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels6, sampler6, t_vertex.m_uv, 0.0)).r) + tide_straight(textureSampleLevel(pixels7, sampler7, t_vertex.m_uv, 0.0)).r) / tide_f(8.0));

    var t_value: f32 = select(select(select(t_d.r, t_c.r, (tide_block.tide_v0.x == 2)), t_b.r, (tide_block.tide_v0.x == 1)), t_a.r, (tide_block.tide_v0.x == 0));
    (*t_color) = vec4f((t_value * t_low), (t_value * t_low), (t_value * t_low), 1.0);

fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_CapacityOne(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec2f = ((t_vertex.m_uv * tide_block.tide_u0.xy) + tide_block.tide_u0.zw);
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);

    (*t_color) = vec4f(t_sampled.r, t_sampled.r, t_sampled.r, 1.0);
}
