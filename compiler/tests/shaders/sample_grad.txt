// Direct on Vertex: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_2d<f32>;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_2d<f32>;
@group(0) @binding(6) var sampler2: sampler;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_Direct(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    let tide_d0: vec2f = dpdx(t_vertex.m_uv);
    let tide_d1: vec2f = dpdy(t_vertex.m_uv);
    let tide_d2: vec2f = dpdx(t_vertex.m_uv);
    let tide_d3: vec2f = dpdy(t_vertex.m_uv);
    var t_dx: vec2f = tide_d0;
    var t_dy: vec2f = tide_d1;
    var t_uv: vec2f = t_vertex.m_uv;
    for (var t_i: i32 = 0; (t_i < 2); t_i = (t_i + 1)) {
        t_uv = (t_uv + vec2f(0.01, 0.0));
    }
    if ((t_vertex.m_uv.x > 0.5)) {
        discard;
        return;
    }
    var tide_choice0: vec2f = t_dy;
    var tide_choice1: vec2f = t_uv;
    var tide_choice2: vec2f = t_dx;
    (*t_color) = tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice1, tide_choice2, tide_choice0));
    var tide_choice3: vec2f = t_uv;
    var tide_choice4: vec2f = t_dx;
    var tide_choice5: vec2f = t_dy;
    (*t_color) = ((*t_color) + tide_straight(textureSampleGrad(pixels1, sampler1, tide_choice3, tide_choice4, tide_choice5)));
    (*t_color) = ((*t_color) + tide_straight(textureSampleLevel(pixels1, sampler1, t_uv, 0.0)));
    var tide_choice6: vec2f = t_vertex.m_uv;
    var tide_choice7: vec2f = t_vertex.m_uv;
    var tide_choice8: vec2f = tide_d2;
    var tide_choice9: vec2f = t_vertex.m_uv;
    var tide_choice10: vec2f = tide_d3;
    (*t_color) = ((*t_color) + tide_straight(textureSampleGrad(pixels2, sampler2, tide_choice6, tide_choice8, tide_choice10)));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Direct(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

@group(0) @binding(9) var pixels4: texture_2d<f32>;
@group(0) @binding(10) var sampler4: sampler;
@group(0) @binding(11) var pixels5: texture_2d<f32>;
var<private> tide_at: vec4f;

fn tf_Gradient_40(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {
    var tide_choice0: vec2f = t_uv;
    var tide_choice1: vec2f = t_dx;
    var tide_choice2: vec2f = t_dy;
    var tide_choice3: vec4f = tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice0, tide_choice1, tide_choice2));
    var tide_choice4: vec2f = t_uv;
    var tide_choice5: vec2f = t_dx;
    var tide_choice6: vec2f = t_dy;
    var tide_choice7: vec4f = tide_straight(textureSampleGrad(pixels1, sampler1, tide_choice4, tide_choice5, tide_choice6));
    return (tide_choice3 + tide_choice7);
}

fn tf_Forward_40(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {
    return tf_Gradient_40(t_uv, t_dx, t_dy);
}

fn tf_Gradient_42(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {

    var tide_choice9: vec2f = t_dx;
    var tide_choice10: vec2f = t_dy;

    var tide_choice12: vec2f = t_uv;
    var tide_choice13: vec2f = t_dx;
    var tide_choice14: vec2f = t_dy;
    var tide_choice15: vec4f = textureSampleGrad(pixels3, sampler1, tide_choice12, tide_choice13, tide_choice14);

}

fn tf_Forward_42(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {
    return tf_Gradient_42(t_uv, t_dx, t_dy);
}

fn tf_Gradient_44(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {
    var tide_choice16: vec2f = t_uv;
    var tide_choice17: vec2f = t_dx;

    var tide_choice19: vec4f = textureSampleGrad(pixels4, sampler4, tide_choice16, tide_choice17, tide_choice18);

fn tf_Forward_44(t_uv: vec2f, t_dx: vec2f, t_dy: vec2f) -> vec4f {

fn ts_Helpers(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    let tide_d0: vec2f = dpdx(t_vertex.m_uv);
    let tide_d1: vec2f = dpdy(t_vertex.m_uv);
    var t_dx: vec2f = tide_d0;
    var t_dy: vec2f = tide_d1;
    (*t_color) = ((tf_Forward_40(t_vertex.m_uv, t_dx, t_dy) + tf_Forward_42(t_vertex.m_uv, t_dx, t_dy)) + tf_Forward_44(t_vertex.m_uv, t_dx, t_dy));
}

// Ordered on Vertex: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Vertex {
    m_position: vec2f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_Next(t_value: ptr<function, vec2f>) -> vec2f {
    (*t_value) = ((*t_value) + vec2f(0.01, 0.02));
    return (*t_value);
}

fn ts_Ordered(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    var t_value: vec2f = t_vertex.m_uv;
    var tide_choice0: vec2f = tf_Next(&t_value);
    var tide_choice1: vec2f = tf_Next(&t_value);
    var tide_choice2: vec2f = tf_Next(&t_value);
    (*t_color) = tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice0, tide_choice1, tide_choice2));
    var tide_choice3: f32 = t_vertex.m_uv.x;
    var tide_choice4: f32 = 0.5;
    var tide_choice5: bool = (tide_choice3 > tide_choice4);
    var tide_choice6: vec4f;
    if (tide_choice5) {
        var tide_choice7: vec2f = tf_Next(&t_value);
        var tide_choice8: vec2f = tf_Next(&t_value);
        var tide_choice9: vec2f = tf_Next(&t_value);
        tide_choice6 = tide_straight(textureSampleGrad(pixels0, sampler0, tide_choice7, tide_choice8, tide_choice9));
    } else {
        tide_choice6 = vec4f(0.0, 0.0, 0.0, 1.0);
    }
    (*t_color) = ((*t_color) + tide_choice6);
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);

}

    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
