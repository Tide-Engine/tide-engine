// Surface on Corner: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform sampler2D pixels0;
uniform sampler2D pixels1;
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec3 tide_a1;
layout(location = 2) in vec2 tide_a2;
layout(location = 3) in vec4 tide_n0;
layout(location = 4) in vec4 tide_n1;
layout(location = 5) in vec4 tide_n2;
layout(location = 6) in vec4 tide_n3;
out vec3 tide_s1;
out vec2 tide_s2;
flat out vec3 tide_normals0;
flat out vec3 tide_normals1;
flat out vec3 tide_normals2;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
    vec2 m_uv;
};
mat3 tide_normals(mat4 m) {
    vec3 a = cross(m[1].xyz, m[2].xyz);
    vec3 b = cross(m[2].xyz, m[0].xyz);
    vec3 c = cross(m[0].xyz, m[1].xyz);
    if (dot(m[0].xyz, a) < 0.0) {
        return mat3(-a, -b, -c);
    }
    return mat3(a, b, c);
}
void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1, tide_a2);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_normal;
    tide_s2 = tide_vertex.m_uv;
    mat3 tide_turned = tide_normals(mat4(tide_n0, tide_n1, tide_n2, tide_n3));
    tide_normals0 = tide_turned[0];
    tide_normals1 = tide_turned[1];
    tide_normals2 = tide_turned[2];
}

// Surface on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform sampler2D pixels0;
uniform sampler2D pixels1;
in vec3 tide_s1;
in vec2 tide_s2;
flat in vec3 tide_normals0;
flat in vec3 tide_normals1;
flat in vec3 tide_normals2;
out vec4 tide_pixel;
struct tt_Corner {
    vec3 m_position;
    vec3 m_normal;
    vec2 m_uv;
};
struct tt_MaterialInput {
    vec3 m_normal;
    vec2 m_uv;
};
vec3 tide_normalize3(vec3 v) {
    float l = dot(v, v);
    if (l > 1.17549435e-38) { return v * inversesqrt(l); }
    return vec3(0.0);
}
vec4 tide_straight(vec4 c) {
    if (c.a == 0.0) { return vec4(0.0); }
    return vec4(c.rgb / c.a, c.a);
}
tt_MaterialInput tf_MakeInput(vec2 t_uv, vec3 t_normal) {
    return tt_MaterialInput(t_normal, t_uv);
}

vec2 tf_MaterialUV(tt_MaterialInput t_input) {
    return (t_input.m_uv + (t_input.m_normal.xy * 0.125));
}

void ts_Surface(tt_Corner t_vertex, inout vec4 t_color) {
    vec2 tide_uv0 = (tf_MaterialUV((tf_MakeInput(t_vertex.m_uv, tide_normalize3(mat3(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal))))));
    vec2 tide_dx0 = dFdx(tide_uv0);
    vec2 tide_dy0 = dFdy(tide_uv0);
    vec2 tide_uv1 = tide_normalize3(mat3(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)).xy;
    vec2 tide_dx1 = dFdx(tide_uv1);
    vec2 tide_dy1 = dFdy(tide_uv1);
    tt_MaterialInput t_frame = tf_MakeInput(t_vertex.m_uv, tide_normalize3(mat3(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)));
    vec2 t_uv = tf_MaterialUV(t_frame);
    if ((t_vertex.m_uv.x > 0.5)) {
        t_color = tide_straight(textureGrad(pixels0, t_uv, tide_dx0, tide_dy0));
    } else {
        t_color = tide_straight(textureGrad(pixels1, tide_normalize3(mat3(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)).xy, tide_dx1, tide_dy1));
    }
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Surface(tt_Corner(vec3(0.0), tide_s1, tide_s2), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Surface on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
@group(0) @binding(3) var pixels1: texture_2d<f32>;
@group(0) @binding(4) var sampler1: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
var<private> tide_normals0: vec3f;
var<private> tide_normals1: vec3f;
var<private> tide_normals2: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec3f,
    @location(1) tide_s2: vec2f,
    @location(2) @interpolate(flat) tide_normals0: vec3f,
    @location(3) @interpolate(flat) tide_normals1: vec3f,
    @location(4) @interpolate(flat) tide_normals2: vec3f,
}
struct tt_Corner {
    m_position: vec3f,
    m_normal: vec3f,
    m_uv: vec2f,
}
struct tt_MaterialInput {
    m_normal: vec3f,
    m_uv: vec2f,
}
fn tide_normalize3(v: vec3f) -> vec3f {
    var l: f32 = dot(v, v);
    if (l > 1.17549435e-38) { return v * inverseSqrt(l); }
    return vec3f(0.0);
}
fn tide_normals(m: mat4x4f) -> mat3x3f {
    var a: vec3f = cross(m[1].xyz, m[2].xyz);
    var b: vec3f = cross(m[2].xyz, m[0].xyz);
    var c: vec3f = cross(m[0].xyz, m[1].xyz);
    if (dot(m[0].xyz, a) < 0.0) {
        return mat3x3f(-a, -b, -c);
    }
    return mat3x3f(a, b, c);
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_MakeInput(t_uv: vec2f, t_normal: vec3f) -> tt_MaterialInput {
    return tt_MaterialInput(t_normal, t_uv);
}

fn tf_MaterialUV(t_input: tt_MaterialInput) -> vec2f {
    return (t_input.m_uv + (t_input.m_normal.xy * 0.125));
}

fn ts_Surface(t_vertex: tt_Corner, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec2f = (tf_MaterialUV((tf_MakeInput(t_vertex.m_uv, tide_normalize3(mat3x3f(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal))))));
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);
    let tide_uv1: vec2f = tide_normalize3(mat3x3f(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)).xy;
    let tide_dx1: vec2f = dpdx(tide_uv1);
    let tide_dy1: vec2f = dpdy(tide_uv1);
    var t_frame: tt_MaterialInput = tf_MakeInput(t_vertex.m_uv, tide_normalize3(mat3x3f(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)));
    var t_uv: vec2f = tf_MaterialUV(t_frame);
    if ((t_vertex.m_uv.x > 0.5)) {
        (*t_color) = tide_straight(textureSampleGrad(pixels0, sampler0, t_uv, tide_dx0, tide_dy0));
    } else {
        (*t_color) = tide_straight(textureSampleGrad(pixels1, sampler1, tide_normalize3(mat3x3f(tide_normals0, tide_normals1, tide_normals2) * (t_vertex.m_normal)).xy, tide_dx1, tide_dy1));
    }
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec3f, @location(2) tide_a2: vec2f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1, tide_a2);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_normal;
    out.tide_s2 = tide_vertex.m_uv;
    let tide_turned = tide_normals(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3));
    out.tide_normals0 = tide_turned[0];
    out.tide_normals1 = tide_turned[1];
    out.tide_normals2 = tide_turned[2];
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_normals0 = in.tide_normals0;
    tide_normals1 = in.tide_normals1;
    tide_normals2 = in.tide_normals2;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Surface(tt_Corner(vec3f(), in.tide_s1, in.tide_s2), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
