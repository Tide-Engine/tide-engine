// Ring on Corner: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform sampler2D pixels0;
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_n0;
layout(location = 3) in vec4 tide_n1;
layout(location = 4) in vec4 tide_n2;
layout(location = 5) in vec4 tide_n3;
out vec2 tide_s1;
out vec3 tide_world;
struct tt_Corner {
    vec3 m_position;
    vec2 m_uv;
};
void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_uv;
    tide_world = tide_place.xyz;
}

// Ring on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
};
uniform sampler2D pixels0;
in vec2 tide_s1;
in vec3 tide_world;
out vec4 tide_pixel;
struct tt_Corner {
    vec3 m_position;
    vec2 m_uv;
};
float tide_f(float x) {
    return x;
}
bool tide_nan(float x) {
    return (floatBitsToInt(x) & 0x7fffffff) > 0x7f800000;
}
float tide_saturate1(float a) {
    if (tide_nan(a)) { return 0.0; }
    return clamp(a, 0.0, 1.0);
}
vec4 tide_straight(vec4 c) {
    if (c.a == 0.0) { return vec4(0.0); }
    return vec4(c.rgb / c.a, c.a);
}
float tf_Disc(vec2 t_uv) {
    return (length((t_uv - vec2(tide_f(0.5)))) - 0.4);
}

void ts_Ring(tt_Corner t_vertex, inout vec4 t_color) {
    float tide_d0 = fwidth((tf_Disc(t_vertex.m_uv)));
    vec2 tide_d1 = dFdx(t_vertex.m_uv);
    vec2 tide_d2 = ((tide_gpu_height > 0 ? 1.0 : -1.0) * dFdy(tide_world.xy));
    vec4 tide_d3 = fwidth(tide_straight(textureLod(pixels0, t_vertex.m_uv, 0.0)));
    float tide_d4 = fwidth(((vec2(gl_FragCoord.x, tide_screen.z - gl_FragCoord.y) / tide_screen.xy).x + ((tide_gpu_height > 0 ? 1.0 : -1.0) * dFdy((tf_Disc(t_vertex.m_uv))))));
    vec3 tide_d5 = fwidth(tide_world);
    float t_d = tf_Disc(t_vertex.m_uv);
    t_color = vec4(1.0, 1.0, 1.0, 1.0);
    if ((t_vertex.m_uv.x > 0.25)) {
        float t_edge = tide_saturate1((tide_f(0.5) - (t_d / tide_d0)));
        t_color = vec4(t_edge, t_edge, t_edge, 1.0);
    }
    vec2 t_slope = (tide_d1 + tide_d2);
    vec4 t_texel = tide_d3;
    for (int t_i = 0; (t_i < 2); t_i = (t_i + 1)) {
        t_color.r = (t_color.r + ((tide_d4 * t_slope.x) * t_texel.a));
    }
    t_color.b = tide_d5.z;
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Ring(tt_Corner(vec3(0.0), tide_s1), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Ring on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
    @location(1) tide_world: vec3f,
}
struct tt_Corner {
    m_position: vec3f,
    m_uv: vec2f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_saturate1(a: f32) -> f32 {
    if (tide_nan(a)) { return 0.0; }
    return clamp(a, 0.0, 1.0);
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_Disc(t_uv: vec2f) -> f32 {
    return (length((t_uv - vec2f(tide_f(0.5)))) - 0.4);
}

fn ts_Ring(t_vertex: tt_Corner, t_color: ptr<function, vec4f>) {
    let tide_d0: f32 = fwidth((tf_Disc(t_vertex.m_uv)));
    let tide_d1: vec2f = dpdx(t_vertex.m_uv);
    let tide_d2: vec2f = dpdy(tide_world.xy);
    let tide_d3: vec4f = fwidth(tide_straight(textureSampleLevel(pixels0, sampler0, t_vertex.m_uv, 0.0)));
    let tide_d4: f32 = fwidth(((tide_at.xy / tide_block.tide_screen.xy).x + dpdy((tf_Disc(t_vertex.m_uv)))));
    let tide_d5: vec3f = fwidth(tide_world);
    var t_d: f32 = tf_Disc(t_vertex.m_uv);
    (*t_color) = vec4f(1.0, 1.0, 1.0, 1.0);
    if ((t_vertex.m_uv.x > 0.25)) {
        var t_edge: f32 = tide_saturate1((tide_f(0.5) - (t_d / tide_d0)));
        (*t_color) = vec4f(t_edge, t_edge, t_edge, 1.0);
    }
    var t_slope: vec2f = (tide_d1 + tide_d2);
    var t_texel: vec4f = tide_d3;
    for (var t_i: i32 = 0; (t_i < 2); t_i = (t_i + 1)) {
        (*t_color).r = ((*t_color).r + ((tide_d4 * t_slope.x) * t_texel.a));
    }
    (*t_color).b = tide_d5.z;
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    out.tide_world = tide_place.xyz;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_world = in.tide_world;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Ring(tt_Corner(vec3f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
