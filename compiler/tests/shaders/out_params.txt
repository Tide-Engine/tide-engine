// Split on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) @invariant tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Corner {
    m_position: vec2f,
    m_uv: vec2f,
}
fn tf_Past(t_value: f32, t_at: f32, t_over: ptr<function, f32>) -> bool {
    if ((t_value < t_at)) {
        (*t_over) = f32();
        return false;
    }
    (*t_over) = (t_value - t_at);
    return true;
}

fn tf_Halves(t_uv: vec2f, t_left: ptr<function, f32>, t_right: ptr<function, f32>) {
    (*t_left) = (t_uv.x * 0.5);
    (*t_right) = (1.0 - (*t_left));
}

fn ts_Split(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(0.0, 0.0, 0.0, 1.0);
    var t_over: f32 = f32();
    if (tf_Past(t_corner.m_uv.x, tide_block.tide_u0.x, &t_over)) {
        (*t_color).r = t_over;
    }
    (*t_color).g = (t_over * 2.0);
    var t_left: f32 = f32();
    var t_tide_0out_043_043: f32 = f32();
    tf_Halves(t_corner.m_uv, &t_left, &t_tide_0out_043_043);
    var t_blue: f32 = 0.0;
    tf_Past(t_left, 0.25, &t_blue);
    (*t_color).b = t_blue;
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f) -> tide_corner {
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Split(tt_Corner(vec2f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
