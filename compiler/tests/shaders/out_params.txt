// Split on Corner: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
};
layout(location = 0) in vec2 tide_a0;
layout(location = 1) in vec2 tide_a1;
out vec2 tide_s1;
struct tt_Corner {
    vec2 m_position;
    vec2 m_uv;
};
void main() {
    tt_Corner tide_vertex = tt_Corner(tide_a0, tide_a1);
    vec4 tide_place = vec4(tide_vertex.m_position, 0.0, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_uv;
}

// Split on Corner: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
};
in vec2 tide_s1;
out vec4 tide_pixel;
struct tt_Corner {
    vec2 m_position;
    vec2 m_uv;
};
bool tf_Past(float t_value, float t_at, inout float t_over) {
    if ((t_value < t_at)) {
        t_over = 0.0;
        return false;
    }
    t_over = (t_value - t_at);
    return true;
}

void tf_Halves(vec2 t_uv, inout float t_left, inout float t_right) {
    t_left = (t_uv.x * 0.5);
    t_right = (1.0 - t_left);
}

void ts_Split(tt_Corner t_corner, inout vec4 t_color) {
    t_color = vec4(0.0, 0.0, 0.0, 1.0);
    float t_over = 0.0;
    if (tf_Past(t_corner.m_uv.x, tide_u0.x, t_over)) {
        t_color.r = t_over;
    }
    t_color.g = (t_over * 2.0);
    float t_left = 0.0;
    float t_tide_0out_043_043 = 0.0;
    tf_Halves(t_corner.m_uv, t_left, t_tide_0out_043_043);
    float t_blue = 0.0;
    tf_Past(t_left, 0.25, t_blue);
    t_color.b = t_blue;
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Split(tt_Corner(vec2(0.0), tide_s1), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Split on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_u0: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Corner {
    m_position: vec2f,
    m_uv: vec2f,
}
fn tf_Past(t_value: f32, t_at: f32, t_over: ptr<function, f32>) -> bool {
    if ((t_value < t_at)) {
        (*t_over) = f32();
        return false;
    }
    (*t_over) = (t_value - t_at);
    return true;
}

fn tf_Halves(t_uv: vec2f, t_left: ptr<function, f32>, t_right: ptr<function, f32>) {
    (*t_left) = (t_uv.x * 0.5);
    (*t_right) = (1.0 - (*t_left));
}

fn ts_Split(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    (*t_color) = vec4f(0.0, 0.0, 0.0, 1.0);
    var t_over: f32 = f32();
    if (tf_Past(t_corner.m_uv.x, tide_block.tide_u0.x, &t_over)) {
        (*t_color).r = t_over;
    }
    (*t_color).g = (t_over * 2.0);
    var t_left: f32 = f32();
    var t_tide_0out_043_043: f32 = f32();
    tf_Halves(t_corner.m_uv, &t_left, &t_tide_0out_043_043);
    var t_blue: f32 = 0.0;
    tf_Past(t_left, 0.25, &t_blue);
    (*t_color).b = t_blue;
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f) -> tide_corner {
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Split(tt_Corner(vec2f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
