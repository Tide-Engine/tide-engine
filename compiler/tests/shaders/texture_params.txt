float tf_Soft_40(vec3 t_seen) {
    float t_lit = textureLod(pixels0, vec3(t_seen.xy, clamp((t_seen.z - 0.01), 0.0, 1.0)), 0.0);
    if ((t_lit < 1.0)) {
        t_lit = ((t_lit + textureLod(pixels0, vec3((t_seen.xy + vec2(0.01, 0.0)), clamp((t_seen.z - 0.01), 0.0, 1.0)), 0.0)) * 0.5);
    }
    return ((textureLod(pixels1, t_seen.xy, 0.0).r < 1.0) ? t_lit : 1.0);
}

vec3 tf_FromLamp(vec3 t_world, mat4 t_toLamp) {
    vec4 t_clip = (t_toLamp * vec4(t_world, tide_f(1.0)));
    return vec3(((t_clip.x * 0.5) + 0.5), (0.5 - (t_clip.y * 0.5)), ((t_clip.z * 0.5) + 0.5));
}

float tf_Sunlight_40(vec3 t_world, mat4 t_toLamp) {
    return tf_Soft_40(tf_FromLamp(t_world, t_toLamp));

    return (0.25 + (0.75 * t_lit));

float tf_Fog_2m0_42(tt_Fog tide_self, vec2 t_uv) {
    return (textureLod(pixels3, t_uv, 0.0).r * tide_self.m_thick);

void ts_Lit(tt_Vertex3 t_vertex, inout vec4 t_color) {
    float t_near = tf_Sunlight_40(tide_world, mat4(tide_u0, tide_u1, tide_u2, tide_u3));
    float t_far = tf_Sunlight_42(tide_world, mat4(tide_u0, tide_u1, tide_u2, tide_u3));

    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
