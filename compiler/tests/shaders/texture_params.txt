struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
    tide_u0: vec4f,
    tide_u1: vec4f,
    tide_u2: vec4f,
    tide_u3: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}

fn tide_f(x: f32) -> f32 {
    return x;
}
fn tf_Soft_40(t_seen: vec3f) -> f32 {
    var t_lit: f32 = textureSampleCompareLevel(pixels0, sampler0, t_seen.xy, clamp((t_seen.z - 0.01), 0.0, 1.0));
    if ((t_lit < 1.0)) {
        t_lit = ((t_lit + textureSampleCompareLevel(pixels0, sampler0, (t_seen.xy + vec2f(0.01, 0.0)), clamp((t_seen.z - 0.01), 0.0, 1.0))) * 0.5);
    }
    return select(1.0, t_lit, (textureSampleLevel(pixels0, sampler1, t_seen.xy, 0) < 1.0));
}

fn tf_FromLamp(t_world: vec3f, t_toLamp: mat4x4f) -> vec3f {
    var t_clip: vec4f = (t_toLamp * vec4f(t_world, tide_f(1.0)));
    return vec3f(((t_clip.x * 0.5) + 0.5), (0.5 - (t_clip.y * 0.5)), ((t_clip.z * 0.5) + 0.5));
}

fn tf_Sunlight_40(t_world: vec3f, t_toLamp: mat4x4f) -> f32 {
    return tf_Soft_40(tf_FromLamp(t_world, t_toLamp));
}

fn tf_Soft_41(t_seen: vec3f) -> f32 {
    var t_lit: f32 = textureSampleCompareLevel(pixels1, sampler0, t_seen.xy, clamp((t_seen.z - 0.01), 0.0, 1.0));
    if ((t_lit < 1.0)) {
        t_lit = ((t_lit + textureSampleCompareLevel(pixels1, sampler0, (t_seen.xy + vec2f(0.01, 0.0)), clamp((t_seen.z - 0.01), 0.0, 1.0))) * 0.5);
    }
    return select(1.0, t_lit, (textureSampleLevel(pixels1, sampler1, t_seen.xy, 0) < 1.0));
}

fn tf_Sunlight_41(t_world: vec3f, t_toLamp: mat4x4f) -> f32 {
    return tf_Soft_41(tf_FromLamp(t_world, t_toLamp));
}

fn tf_Ambient_40(t_lit: f32) -> f32 {
    return (0.25 + (0.75 * t_lit));
}

fn tf_Fog_2m0_41(tide_self: tt_Fog, t_uv: vec2f) -> f32 {
    return (textureSampleLevel(pixels1, sampler1, t_uv, 0) * tide_self.m_thick);
}

fn ts_Lit(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    var t_near: f32 = tf_Sunlight_40(tide_world, mat4x4f(tide_block.tide_u0, tide_block.tide_u1, tide_block.tide_u2, tide_block.tide_u3));
    var t_far: f32 = tf_Sunlight_41(tide_world, mat4x4f(tide_block.tide_u0, tide_block.tide_u1, tide_block.tide_u2, tide_block.tide_u3));
    var t_fog: tt_Fog = tt_Fog(0.5);
    (*t_color) = ((vec4f(1.0, 1.0, 1.0, 1.0) * tf_Ambient_40((t_near * t_far))) * (1.0 - tf_Fog_2m0_41(t_fog, t_vertex.m_uv)));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    out.tide_world = tide_place.xyz;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_world = in.tide_world;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Vertex3(vec3f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

fn tide_list3(tide_i: vec2i) -> vec4f {
    let tide_o = tide_block.tide_v2.xy;
    let tide_e = tide_block.tide_v2.zw;
    let tide_x = u32(tide_i.x) - u32(tide_o.x);
    let tide_y = u32(tide_i.y) - u32(tide_o.y);
    if (tide_i.x < tide_o.x || tide_x >= u32(tide_e.x) || tide_i.y < tide_o.y || tide_y >= u32(tide_e.y)) {
        return vec4f();
    }
    let tide_word = i32(tide_x) * 4;
    let tide_first = tide_word >> 2u;
    let tide_row = i32(tide_y);
    let tide_q0 = textureLoad(pixels3, vec2i(tide_first + 0, tide_row), 0);
    let tide_p0 = bitcast<vec4f>(tide_q0);
    return tide_p0;
}

fn tf_Plank_43(t_uv: vec2f) -> vec4f {
    return (tide_straight(textureSampleLevel(pixels4, sampler3, t_uv, 0.0)) + tide_list3(vec2i(0, 0)));
}

fn ts_Shine(t_vertex: tt_Vertex, t_color: ptr<function, vec4f>) {
    (*t_color) = (tf_Mirror_40(vec3f(t_vertex.m_uv, 1.0)) + tf_Mirror_41(vec3f(t_vertex.m_uv, f32((-1)))));
}

@vertex fn vertex(@location(0) tide_a0: vec2f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f) -> tide_corner {
    var tide_vertex: tt_Vertex = tt_Vertex(tide_a0, tide_a1, tide_a2);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Shine(tt_Vertex(vec2f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
