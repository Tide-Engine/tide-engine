// Leaf on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
var<private> tide_cover: f32;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Corner {
    m_position: vec3f,
    m_uv: vec2f,
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_Leaf(t_vertex: tt_Corner, t_color: ptr<function, vec4f>) {
    let tide_uv0: vec2f = t_vertex.m_uv;
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);
    (*t_color) = tide_straight(textureSampleGrad(pixels0, sampler0, t_vertex.m_uv, tide_dx0, tide_dy0));
    tide_cover = (*t_color).a;
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    tide_cover = 1.0;
    ts_Leaf(tt_Corner(vec3f(), in.tide_s1), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    var tide_share: f32 = tide_cover * tide_block.tide_tint.a;
    tide_share = (tide_share - 0.5) / max(fwidth(tide_share), 0.0001) + 0.5;
    if (tide_block.tide_screen.w < 0.0) {
        if (tide_share < 0.5) {
            discard;
        }
        tide_share = 1.0;
    }
    tide_color = vec4f(tide_color.rgb, clamp(tide_share, 0.0, 1.0));
    return tide_color;
}

// Glowing on Corner: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_origin: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
fn tide_project(model: mat4x4f, vertex: vec4f) -> vec4f {
    let relative = mat4x4f(
        model[0] - tide_block.tide_origin * model[0].w,
        model[1] - tide_block.tide_origin * model[1].w,
        model[2] - tide_block.tide_origin * model[2].w,
        model[3] - tide_block.tide_origin * model[3].w);
    return tide_block.tide_camera * (relative * vertex);
}
@group(0) @binding(1) var pixels0: texture_2d<f32>;
@group(0) @binding(2) var sampler0: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
var<private> tide_cover: f32;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
}
struct tt_Corner {
    m_position: vec3f,
    m_uv: vec2f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_nan(x: f32) -> bool {
    return (bitcast<i32>(x) & 0x7fffffff) > 0x7f800000;
}
fn tide_saturate1(a: f32) -> f32 {
    if (tide_nan(a)) { return 0.0; }
    return clamp(a, 0.0, 1.0);
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn ts_Glowing(t_vertex: tt_Corner, t_color: ptr<function, vec4f>, t_glow: ptr<function, vec4f>) {
    let tide_uv0: vec2f = t_vertex.m_uv;
    let tide_dx0: vec2f = dpdx(tide_uv0);
    let tide_dy0: vec2f = dpdy(tide_uv0);
    let tide_d1: f32 = fwidth((tide_straight(textureSample(pixels0, sampler0, t_vertex.m_uv)).a));
    if ((t_vertex.m_uv.y > 0.9)) {
        discard;
        return;
    }
    var t_alpha: f32 = tide_straight(textureSampleGrad(pixels0, sampler0, t_vertex.m_uv, tide_dx0, tide_dy0)).a;
    (*t_color) = vec4f(0.0, 1.0, 0.0, 1.0);
    if ((t_vertex.m_uv.x > 0.5)) {
        tide_cover = tide_saturate1((((t_alpha - tide_f(0.3)) / tide_d1) + tide_f(0.5)));
    }
    (*t_glow) = vec4f(1.0, 0.5, 0.0, 0.5);
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    let tide_projected = tide_project(mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3), vec4f(tide_vertex.m_position, 1.0));
    let tide_clip = tide_projected;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    return out;
}
struct tide_colors {
    @location(0) tide_color: vec4f,
    @location(1) tide_color1: vec4f,
}
@fragment fn fragment(in: tide_corner) -> tide_colors {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    var tide_color1: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    tide_cover = 1.0;
    ts_Glowing(tt_Corner(vec3f(), in.tide_s1), &tide_color, &tide_color1);
    tide_color = tide_color * tide_block.tide_tint;
    var tide_share: f32 = tide_cover * tide_block.tide_tint.a;
    if (tide_block.tide_screen.w < 0.0) {
        if (tide_share < 0.5) {
            discard;
        }
        tide_share = 1.0;
    }
    tide_color = vec4f(tide_color.rgb, clamp(tide_share, 0.0, 1.0));
    tide_color1 = tide_color1 * tide_block.tide_tint;
    tide_color1 = vec4f(tide_color1.rgb * tide_color1.a, tide_color1.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color1 = vec4f(tide_color1.rgb + (1.0 - tide_color1.a), tide_color1.a);
    }
    return tide_colors(tide_color, tide_color1);
}
