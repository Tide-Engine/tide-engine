@vertex fn vertex(@location(0) tide_a0: vec2f) -> tide_corner {
    var tide_vertex: tt_Corner = tt_Corner(tide_a0);
    let tide_place = vec4f(tide_vertex.m_position, 0.0, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_ShowWide(tt_Corner(vec2f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}

fn ts_Spread() {
    var t_i: i32 = tide_inv_index;
    var t_x: f32 = (tide_get0((t_i % tide_count0())) + tide_get1(0));
    var t_n: i32 = tide_get2((t_i % tide_i(4)));
    tide_put3(t_i, (f32(t_i) + t_x));
    tide_put4(t_i, (f32(t_i) * 2.0));
    tide_put5(t_i, (f32(t_i) * 3.0));
    tide_put6(t_i, (f32(t_i) * 4.0));
    tide_put7(t_i, (f32(t_i) * 5.0));
    tide_put8(t_i, (f32(t_i) * 6.0));
    _ = tide_atomic9_AtomicAdd((t_i % tide_i(4)), 1);
    _ = tide_atomic10_AtomicAdd((t_i % tide_i(4)), t_n);
    _ = tide_atomic11_AtomicMax(0, t_i);
    _ = tide_atomic12_AtomicOr(0, (1 << u32((t_i % tide_i(31)))));
    if (((t_i % tide_i(2)) == 0)) {
        tide_adding0 = true;
        tide_added0 = t_i;
    }
    if (((t_i % tide_i(3)) == 0)) {
        tide_adding1 = true;
        tide_added1 = (t_i * 2);
    }
    if (((t_i % tide_i(5)) == 0)) {
        tide_adding2 = true;
        tide_added2 = (t_i + t_n);
    }
    if (((t_i % tide_i(7)) == 0)) {
        tide_adding3 = true;
        tide_added3 = (-t_i);
    }
    if (((t_i % tide_i(11)) == 0)) {
        tide_adding4 = true;
        tide_added4 = t_i;
    }
    if (((t_i % tide_i(4)) == 1)) {
        tide_adding5 = true;
        tide_added5 = vec2f(f32(t_i), t_x);
    }
    tide_put19(t_i, tt_Item(t_x, t_n, vec2f(f32(t_i), 1.0), ((t_i % tide_i(2)) == 0)));
}

    tide_scan[tide_lix] = vec4u(select(0u, 1u, tide_adding0), select(0u, 1u, tide_adding1), select(0u, 1u, tide_adding2), select(0u, 1u, tide_adding3));
    tide_scan1[tide_lix] = vec2u(select(0u, 1u, tide_adding4), select(0u, 1u, tide_adding5));
    workgroupBarrier();
    for (var tide_d = 1u; tide_d < 64u; tide_d = tide_d * 2u) {

        var tide_v1 = vec2u(0u);
        if (tide_lix >= tide_d) {
            tide_v = tide_scan[tide_lix - tide_d];
            tide_v1 = tide_scan1[tide_lix - tide_d];
        }
        workgroupBarrier();
        tide_scan[tide_lix] = tide_scan[tide_lix] + tide_v;
        tide_scan1[tide_lix] = tide_scan1[tide_lix] + tide_v1;

    }
    let tide_region0 = u32(tide_block.tide_v10.x) + (tide_wid.x + tide_groups.x * (tide_wid.y + tide_groups.y * tide_wid.z)) * 66u;
