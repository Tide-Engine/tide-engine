// Lit on Vertex3: GLSL, vertex
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
    vec4 tide_u1;
    vec4 tide_u2;
    vec4 tide_u3;
};
uniform highp sampler2DShadow pixels0;
uniform highp sampler2D pixels1;
uniform sampler2D pixels2;
layout(location = 0) in vec3 tide_a0;
layout(location = 1) in vec2 tide_a1;
layout(location = 2) in vec4 tide_a2;
layout(location = 3) in vec4 tide_n0;
layout(location = 4) in vec4 tide_n1;
layout(location = 5) in vec4 tide_n2;
layout(location = 6) in vec4 tide_n3;
out vec2 tide_s1;
out vec3 tide_world;
struct tt_Vertex3 {
    vec3 m_position;
    vec2 m_uv;
    vec4 m_color;
};
void main() {
    tt_Vertex3 tide_vertex = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    vec4 tide_place = mat4(tide_n0, tide_n1, tide_n2, tide_n3) * vec4(tide_vertex.m_position, 1.0);
    gl_Position = tide_camera * tide_place;
    tide_s1 = tide_vertex.m_uv;
    tide_world = tide_place.xyz;
}

// Lit on Vertex3: GLSL, fragment
layout(std140) uniform uniforms {
    mat4 tide_camera;
    vec4 tide_screen;
    vec4 tide_tint;
    vec4 tide_u0;
    vec4 tide_u1;
    vec4 tide_u2;
    vec4 tide_u3;
};
uniform highp sampler2DShadow pixels0;
uniform highp sampler2D pixels1;
uniform sampler2D pixels2;
in vec2 tide_s1;
in vec3 tide_world;
out vec4 tide_pixel;
struct tt_Vertex3 {
    vec3 m_position;
    vec2 m_uv;
    vec4 m_color;
};
float tide_f(float x) {
    return x;
}
vec4 tide_straight(vec4 c) {
    if (c.a == 0.0) { return vec4(0.0); }
    return vec4(c.rgb / c.a, c.a);
}
vec3 tf_FromLamp(vec3 t_world, mat4 t_toLamp) {
    vec4 t_clip = (t_toLamp * vec4(t_world, tide_f(1.0)));
    return vec3(((t_clip.x * 0.5) + 0.5), (0.5 - (t_clip.y * 0.5)), ((t_clip.z * 0.5) + 0.5));
}

void ts_Lit(tt_Vertex3 t_vertex, inout vec4 t_color) {
    vec3 t_at = tf_FromLamp(tide_world, mat4(tide_u0, tide_u1, tide_u2, tide_u3));
    float t_lit = textureGrad(pixels0, vec3(t_at.xy, clamp((t_at.z - 0.01), 0.0, 1.0)), vec2(0.0), vec2(0.0));
    if ((t_lit < 1.0)) {
        t_lit = ((t_lit * 0.5) + (textureGrad(pixels0, vec3((t_at.xy + vec2(0.01, 0.0)), clamp((t_at.z - 0.01), 0.0, 1.0)), vec2(0.0), vec2(0.0)) * 0.5));
    }
    float t_far = textureLod(pixels1, t_at.xy, 0.0).r;
    t_color = ((tide_straight(textureLod(pixels2, t_vertex.m_uv, 0.0)) * (0.25 + (0.75 * t_lit))) * ((t_far < 1.0) ? 1.0 : 0.5));
}

void main() {
    vec4 tide_color = vec4(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Vertex3(vec3(0.0), tide_s1, vec4(0.0)), tide_color);
    tide_color = tide_color * tide_tint;
    tide_color.rgb = tide_color.rgb * tide_color.a;
    if (tide_screen.w > 0.0) tide_color.rgb = tide_color.rgb + (1.0 - tide_color.a);
    tide_pixel = tide_color;
}

// Lit on Vertex3: WGSL
struct tide_uniforms {
    tide_camera: mat4x4f,
    tide_screen: vec4f,
    tide_tint: vec4f,
    tide_u0: vec4f,
    tide_u1: vec4f,
    tide_u2: vec4f,
    tide_u3: vec4f,
}
@group(0) @binding(0) var<uniform> tide_block: tide_uniforms;
@group(0) @binding(1) var pixels0: texture_depth_2d;
@group(0) @binding(2) var sampler0: sampler_comparison;
@group(0) @binding(3) var pixels1: texture_depth_2d;
@group(0) @binding(4) var sampler1: sampler;
@group(0) @binding(5) var pixels2: texture_2d<f32>;
@group(0) @binding(6) var sampler2: sampler;
var<private> tide_n0: vec4f;
var<private> tide_n1: vec4f;
var<private> tide_n2: vec4f;
var<private> tide_n3: vec4f;
var<private> tide_at: vec4f;
var<private> tide_world: vec3f;
struct tide_corner {
    @builtin(position) tide_at: vec4f,
    @location(0) tide_s1: vec2f,
    @location(1) tide_world: vec3f,
}
struct tt_Vertex3 {
    m_position: vec3f,
    m_uv: vec2f,
    m_color: vec4f,
}
fn tide_f(x: f32) -> f32 {
    return x;
}
fn tide_straight(c: vec4f) -> vec4f {
    if (c.a == 0.0) { return vec4f(0.0); }
    return vec4f(c.rgb / c.a, c.a);
}
fn tf_FromLamp(t_world: vec3f, t_toLamp: mat4x4f) -> vec3f {
    var t_clip: vec4f = (t_toLamp * vec4f(t_world, tide_f(1.0)));
    return vec3f(((t_clip.x * 0.5) + 0.5), (0.5 - (t_clip.y * 0.5)), ((t_clip.z * 0.5) + 0.5));
}

fn ts_Lit(t_vertex: tt_Vertex3, t_color: ptr<function, vec4f>) {
    var t_at: vec3f = tf_FromLamp(tide_world, mat4x4f(tide_block.tide_u0, tide_block.tide_u1, tide_block.tide_u2, tide_block.tide_u3));
    var t_lit: f32 = textureSampleCompareLevel(pixels0, sampler0, t_at.xy, clamp((t_at.z - 0.01), 0.0, 1.0));
    if ((t_lit < 1.0)) {
        t_lit = ((t_lit * 0.5) + (textureSampleCompareLevel(pixels0, sampler0, (t_at.xy + vec2f(0.01, 0.0)), clamp((t_at.z - 0.01), 0.0, 1.0)) * 0.5));
    }
    var t_far: f32 = textureSampleLevel(pixels1, sampler1, t_at.xy, 0);
    (*t_color) = ((tide_straight(textureSampleLevel(pixels2, sampler2, t_vertex.m_uv, 0.0)) * (0.25 + (0.75 * t_lit))) * select(0.5, 1.0, (t_far < 1.0)));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_a2: vec4f, @location(3) tide_in0: vec4f, @location(4) tide_in1: vec4f, @location(5) tide_in2: vec4f, @location(6) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Vertex3 = tt_Vertex3(tide_a0, tide_a1, tide_a2);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    out.tide_s1 = tide_vertex.m_uv;
    out.tide_world = tide_place.xyz;
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    tide_world = in.tide_world;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Vertex3(vec3f(), in.tide_s1, vec4f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
