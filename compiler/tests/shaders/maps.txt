    let tide_base = ((tide_block.tide_v1.y * 8 + 3) >> 2u) << 2u;
    var tide_s = tide_hash1(tide_key) & (tide_slots - 1);
    for (var tide_n: i32 = 0; tide_n < tide_slots; tide_n++) {
        let tide_w = tide_base + tide_s;
        let tide_slot = textureLoad(pixels2, vec2i((tide_w >> 2u) & 1023, tide_w >> 12u), 0);
        let tide_e = tide_slot[tide_w & 3];
        if (tide_e == 0) {
            return -1;
        }
        let tide_k = (tide_e - 1) * 8 + 4;
        let tide_held = textureLoad(pixels2, vec2i((tide_k >> 2u) & 1023, tide_k >> 12u), 0);
        if (tide_held[tide_k & 3] == tide_key) {
            return tide_e - 1;
        }
        tide_s = (tide_s + 1) & (tide_slots - 1);
    }
    return -1;
}

fn tide_try0(tide_key: i32, tide_value: ptr<function, tt_Tile>) -> bool {
    let tide_e = tide_find0(tide_key);
    if (tide_e < 0) {
        *tide_value = tt_Tile(vec4f(0.5, 0.5, 0.5, 1.0), f32(), 1);
        return false;
    }
    *tide_value = tide_list0(tide_e).m_value;
    return true;
}

fn tide_list3(tide_i: i32) -> i32 {
    if (tide_i < 0 || tide_i >= tide_block.tide_v1.w) {
        return i32();
    }
    let tide_word = tide_i * 1;
    let tide_first = tide_word >> 2u;
    let tide_r = textureLoad(pixels3, vec2i((tide_first + 0) & 1023, (tide_first + 0) >> 10u), 0);
    let tide_q0 = vec4i(tide_r[tide_word & 3], 0, 0, 0);
    return tide_q0.x;
}

fn tide_list4(tide_i: i32) -> tt_Entry_4byte_6float_5 {
    if (tide_i < 0 || tide_i >= tide_block.tide_v2.y) {
        return tt_Entry_4byte_6float_5();
    }
    let tide_word = tide_i * 2;
    let tide_first = tide_word >> 2u;
    let tide_r = textureLoad(pixels4, vec2i((tide_first + 0) & 1023, (tide_first + 0) >> 10u), 0);
    let tide_q0 = vec4i(tide_r[tide_word & 3], tide_r[(tide_word & 3) + 1], 0, 0);
    let tide_p0 = bitcast<vec4f>(tide_q0);
    return tt_Entry_4byte_6float_5(tide_q0.y, tide_p0.x);
}

fn tide_find4(tide_key: i32) -> i32 {
    let tide_slots = tide_block.tide_v2.z;
    if (tide_slots <= 0) {
        return -1;
    }
    let tide_base = ((tide_block.tide_v2.y * 2 + 3) >> 2u) << 2u;
    var tide_s = tide_hash1(tide_key) & (tide_slots - 1);
    for (var tide_n: i32 = 0; tide_n < tide_slots; tide_n++) {
        let tide_w = tide_base + tide_s;
        let tide_slot = textureLoad(pixels4, vec2i((tide_w >> 2u) & 1023, tide_w >> 12u), 0);
        let tide_e = tide_slot[tide_w & 3];
        if (tide_e == 0) {
            return -1;
        }
        let tide_k = (tide_e - 1) * 2 + 1;
        let tide_held = textureLoad(pixels4, vec2i((tide_k >> 2u) & 1023, tide_k >> 12u), 0);
        if (tide_held[tide_k & 3] == tide_key) {
            return tide_e - 1;
        }
        tide_s = (tide_s + 1) & (tide_slots - 1);
    }
    return -1;
}

fn ts_Lit(t_corner: tt_Corner, t_color: ptr<function, vec4f>) {
    var t_sum: vec4f = tide_list2(tide_find2(tide_block.tide_v0.x)).m_value;
    var t_water: tt_Tile = tt_Tile(vec4f(0.5, 0.5, 0.5, 1.0), f32(), 1);
    if (tide_try0(2, &t_water)) {
        t_sum = (t_sum + (t_water.m_tint * f32(t_water.m_cost)));
    }
    for (var tide_k0: i32 = 0; tide_k0 < tide_block.tide_v0.y; tide_k0++) {
        var t_entry: tt_Entry_4Ground_6Tile_5 = tide_list0(tide_k0);
        if ((t_entry.m_key == 0)) {
            continue;
        }
        t_sum.g = (t_sum.g + t_entry.m_value.m_height);
    }
    for (var tide_k1: i32 = 0; tide_k1 < tide_block.tide_v1.w; tide_k1++) {
        var t_ground: i32 = tide_list3(tide_k1);
        if ((t_ground == 1)) {
            t_sum.r = (t_sum.r + 0.25);
        }
    }
    var t_found: tt_Tile = tt_Tile(vec4f(0.5, 0.5, 0.5, 1.0), f32(), 1);
    var t_there: bool = tide_try0(0, &t_found);
    t_sum.b = (t_sum.b + (((tide_list4(tide_find4((i32(tide_block.tide_v0.x) & 255))).m_value * f32(tide_block.tide_v1.y)) + f32(tide_block.tide_v0.w)) + f32(tide_block.tide_v1.w)));
    (*t_color) = select((t_sum * t_found.m_height), t_sum, (t_there && (tide_find2(7) >= 0)));
}

@vertex fn vertex(@location(0) tide_a0: vec3f, @location(1) tide_a1: vec2f, @location(2) tide_in0: vec4f, @location(3) tide_in1: vec4f, @location(4) tide_in2: vec4f, @location(5) tide_in3: vec4f) -> tide_corner {
    tide_n0 = tide_in0;
    tide_n1 = tide_in1;
    tide_n2 = tide_in2;
    tide_n3 = tide_in3;
    var tide_vertex: tt_Corner = tt_Corner(tide_a0, tide_a1);
    ts_Raise(&tide_vertex);
    let tide_place = mat4x4f(tide_n0, tide_n1, tide_n2, tide_n3) * vec4f(tide_vertex.m_position, 1.0);
    let tide_clip = tide_block.tide_camera * tide_place;
    var out: tide_corner;
    out.tide_at = vec4f(tide_clip.xy, (tide_clip.z + tide_clip.w) * 0.5, tide_clip.w);
    return out;
}
@fragment fn fragment(in: tide_corner) -> @location(0) vec4f {
    tide_at = in.tide_at;
    var tide_color: vec4f = vec4f(1.0, 1.0, 1.0, 1.0);
    ts_Lit(tt_Corner(vec3f(), vec2f()), &tide_color);
    tide_color = tide_color * tide_block.tide_tint;
    tide_color = vec4f(tide_color.rgb * tide_color.a, tide_color.a);
    if (tide_block.tide_screen.w > 0.0) {
        tide_color = vec4f(tide_color.rgb + (1.0 - tide_color.a), tide_color.a);
    }
    return tide_color;
}
