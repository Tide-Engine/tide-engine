settings { title = "Dodge"; }

singleton Run
{
    float time;
    float spawnTimer;
    bool over;
    bool clear;
    float finalTime;
}

component Body
{
    [Position] float2 position;
    float2 size = float2(24, 24);
}

component Player { }
component Hazard { }

input Controls
{
    float move;
    bool restart;

    Sample(Devices devices)
    {
        var keys = devices.keyboard;
        if (keys.d.pressed || keys.rightArrow.pressed) move += 1;
        if (keys.a.pressed || keys.leftArrow.pressed) move -= 1;
        restart = keys.space.pressed;
    }
}

scene Main { }

event(PlayerJoined joined) AddPlayer()
{
    Spawn(Body { position = float2(0, -150) }, Player, Owner { player = joined.player });
}

system MovePlayer(Controls input, mut Body body, Run run, Time time, with Player, with Owner)
{
    if (run.over) return;
    body.position.x = Math.Clamp(body.position.x + input.move * 300 * time.dt, -320 + 12, 320 - 12);
}

system Clock(mut Run run, Time time)
{
    if (run.over) return;
    run.time += time.dt;
    run.spawnTimer += time.dt;
    if (run.spawnTimer >= 0.5)
    {
        run.spawnTimer -= 0.5;
        Spawn(Body { position = float2(Random.NextFloat(-308, 308), 200), size = float2(20, 20) }, Hazard);
    }
}

system FallHazards(mut Body body, Run run, Time time, with Hazard)
{
    if (run.over) return;
    body.position.y -= (100 + run.time * 12) * time.dt;
    if (body.position.y < -200) this.Destroy();
}

system Collide(Body body, mut Run run, query (Body h, with Hazard) hazards, with Player)
{
    if (run.over) return;
    foreach (var hz in hazards)
    {
        var half = (body.size + hz.h.size) * 0.5;
        var d = body.position - hz.h.position;
        if (Math.Abs(d.x) < half.x && Math.Abs(d.y) < half.y)
        {
            run.over = true;
            run.finalTime = run.time;
        }
    }
}

system Restart(Controls input, mut Run run, with Player, with Owner)
{
    if (!run.over || !input.restart.down) return;
    run.over = false;
    run.time = 0;
    run.spawnTimer = 0;
    run.clear = true;
}

system ClearHazards(Run run, with Hazard)
{
    if (run.clear) this.Destroy();
}

system ResetPlayer(Run run, mut Body body, with Player)
{
    if (run.clear) body.position.x = 0;
}

system EndClear(mut Run run)
{
    run.clear = false;
}

view Render()
{
    Draw.SetCamera(float2(0, 0), Math.Max(180, 320 * Screen.height / Screen.width));
    Draw.Rect(float2(0, 0), float2(640, 360), Color.black);
}

view DrawBodies(Body body, Run run, with Player)
{
    Draw.Rect(body.position, body.size, run.over ? Color.gray : Color.green);
}

view DrawHazards(Body body, with Hazard)
{
    Draw.Rect(body.position, body.size, Color.red);
}

view Hud(Run run)
{
    if (run.over)
    {
        Draw.Text("Game over", float2(0, 30), 32, Color.white, Anchor.MiddleCenter);
        Draw.Text($"You survived {run.finalTime:F1} seconds", float2(0, -5), 20, Color.white, Anchor.MiddleCenter);
        Draw.Text("Press Space to restart", float2(0, -35), 16, Color.gray, Anchor.MiddleCenter);
    }
    else
    {
        Draw.Text($"Time: {run.time:F1}", float2(-300, 165), 16, Color.white, Anchor.MiddleLeft);
    }
}
