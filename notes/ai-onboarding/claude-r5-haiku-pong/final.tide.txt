// Two-player Pong on a 640 x 360 court, in world units (one unit per pixel).
// The first player to join is the left paddle, the second the right one.
settings { title = "Pong"; tickRate = 60; maxPlayers = 2; }

const float HALF_WIDTH = 320;
const float HALF_HEIGHT = 180;
const float PADDLE_X = 300;
const float2 PADDLE_SIZE = float2(8, 56);
const float PADDLE_SPEED = 240;
const float BALL_RADIUS = 4;
const float BALL_SPEED = 240;
const int WIN_SCORE = 5;
const float WIN_SECONDS = 3;

component Body
{
    float2 position;
    float2 velocity;
}

component Ball { }

component Paddle
{
    float2 position;
    float side; // -1 for the left paddle, 1 for the right one
}

// The seats and the score, shared by the whole match.
singleton Court
{
    bool hasLeft;
    bool hasRight;
    PlayerID leftPlayer;
    PlayerID rightPlayer;
    int leftScore;
    int rightScore;
    int winner;          // 0 while playing, otherwise -1 (left) or 1 (right)
    float winTimer;      // seconds left on the win message
    float serveSide = -1; // the side the next serve goes to: the player who conceded
}

input Controls
{
    float2 move;

    Sample(Devices devices)
    {
        var keys = devices.keyboard;
        if (keys.w.pressed || keys.upArrow.pressed) move.y += 1;
        if (keys.s.pressed || keys.downArrow.pressed) move.y -= 1;
    }
}

scene Main { }

float2 ServeVelocity(float side)
{
    return float2(side * BALL_SPEED, 60);
}

event(Spawned) Setup(with Main)
{
    Spawn(Body { }, Ball);
}

event(PlayerJoined joined) Join(mut Court court)
{
    if (!court.hasLeft)
    {
        court.hasLeft = true;
        court.leftPlayer = joined.player;
        Spawn(Paddle { position = float2(-PADDLE_X, 0), side = -1 }, Owner { player = joined.player });
    }
    else if (!court.hasRight)
    {
        court.hasRight = true;
        court.rightPlayer = joined.player;
        Spawn(Paddle { position = float2(PADDLE_X, 0), side = 1 }, Owner { player = joined.player });
    }
}

event(PlayerLeft left) Leave(mut Court court, query (Owner owner, with Paddle) paddles)
{
    if (!court.hasLeft && !court.hasRight) return;
    if (court.hasLeft && court.leftPlayer == left.player) court.hasLeft = false;
    if (court.hasRight && court.rightPlayer == left.player) court.hasRight = false;
    foreach (var row in paddles)
    {
        if (row.owner.player == left.player) row.entity.Destroy();
    }
    // A match with a seat empty starts over when it's full again.
    court.leftScore = 0;
    court.rightScore = 0;
    court.winner = 0;
    court.serveSide = -1;
}

system Steer(Controls input, Time time, mut Paddle paddle, with Owner)
{
    var y = paddle.position.y + input.move.y * PADDLE_SPEED * time.dt;
    paddle.position.y = Math.Clamp(y, -(HALF_HEIGHT - PADDLE_SIZE.y / 2), HALF_HEIGHT - PADDLE_SIZE.y / 2);
}

system Play(Time time, mut Court court, mut Body ball, with Ball, query (Paddle paddle) paddles)
{
    // Waiting for a second player: the ball stays in the center.
    if (!court.hasLeft || !court.hasRight)
    {
        ball.position = float2(0, 0);
        ball.velocity = float2(0, 0);
        return;
    }
    // The win message shows for a while, then the scores reset and the ball serves again.
    if (court.winner != 0)
    {
        court.winTimer -= time.dt;
        if (court.winTimer > 0) return;
        court.winner = 0;
        court.leftScore = 0;
        court.rightScore = 0;
        ball.position = float2(0, 0);
        ball.velocity = ServeVelocity(court.serveSide);
        return;
    }
    // Two players joined: the first serve.
    if (Math.Length(ball.velocity) == 0) ball.velocity = ServeVelocity(court.serveSide);

    ball.position += ball.velocity * time.dt;

    if (ball.position.y + BALL_RADIUS > HALF_HEIGHT || ball.position.y - BALL_RADIUS < -HALF_HEIGHT)
    {
        ball.velocity.y = -ball.velocity.y;
        ball.position.y = Math.Clamp(ball.position.y, -(HALF_HEIGHT - BALL_RADIUS), HALF_HEIGHT - BALL_RADIUS);
    }

    foreach (var row in paddles)
    {
        var paddle = row.paddle;
        var toward = ball.velocity.x * paddle.side > 0;
        var inReach = Math.Abs(ball.position.x - paddle.position.x) <= PADDLE_SIZE.x / 2 + BALL_RADIUS;
        var inLine = Math.Abs(ball.position.y - paddle.position.y) <= PADDLE_SIZE.y / 2 + BALL_RADIUS;
        if (toward && inReach && inLine)
        {
            // Bounce off the paddle's outer face, angled by where the ball hit it.
            ball.position.x = paddle.position.x - paddle.side * (PADDLE_SIZE.x / 2 + BALL_RADIUS);
            ball.velocity.x = -ball.velocity.x;
            ball.velocity.y = (ball.position.y - paddle.position.y) * 4;
        }
    }

    var leftOut = ball.position.x < -HALF_WIDTH - BALL_RADIUS;
    var rightOut = ball.position.x > HALF_WIDTH + BALL_RADIUS;
    if (leftOut || rightOut)
    {
        // The player who missed concedes, and the ball serves back toward them.
        court.serveSide = leftOut ? -1 : 1;
        if (leftOut) court.rightScore += 1;
        else court.leftScore += 1;

        ball.position = float2(0, 0);
        if (court.leftScore == WIN_SCORE || court.rightScore == WIN_SCORE)
        {
            court.winner = court.leftScore == WIN_SCORE ? -1 : 1;
            court.winTimer = WIN_SECONDS;
            ball.velocity = float2(0, 0);
        }
        else
        {
            ball.velocity = ServeVelocity(court.serveSide);
        }
    }
}

view Field(Court court)
{
    // Keep the whole 640 x 360 court in view, widening it for a narrow window.
    Draw.SetCamera(float2(0, 0), Math.Max(180, 320 * Screen.height / Screen.width));
    Draw.Clear(Color(0.05, 0.05, 0.07));

    for (var i = 0; i < 9; i++)
        Draw.Rect(float2(0, 160 - i * 40), float2(4, 20), Color(0.3, 0.3, 0.35));

    Draw.Text($"{court.leftScore}", float2(-80, 150), 48, Color.white, Anchor.MiddleCenter);
    Draw.Text($"{court.rightScore}", float2(80, 150), 48, Color.white, Anchor.MiddleCenter);

    if (court.winner != 0)
    {
        var message = court.winner < 0 ? "LEFT PLAYER WINS" : "RIGHT PLAYER WINS";
        Draw.Text(message, float2(0, 60), 32, Color.yellow, Anchor.MiddleCenter);
    }
    else if (!court.hasLeft || !court.hasRight)
    {
        Draw.Text("Waiting for a second player", float2(0, -150), 16, Color.gray, Anchor.MiddleCenter);
    }
}

view DrawPaddles(Paddle paddle)
{
    Draw.Rect(paddle.position, PADDLE_SIZE, Color.white);
}

view DrawBall(Body ball, with Ball)
{
    Draw.Circle(ball.position, BALL_RADIUS, Color.white);
}
