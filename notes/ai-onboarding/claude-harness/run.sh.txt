#!/bin/sh
# usage: run.sh <trial-name> <model> <task-file>
set -eu
ROOT=/c/Users/blenm/tide-ai-trials
T="$ROOT/trials/$1"
export TIDE_HIDDEN=1
export PATH="$ROOT/sdk/bin:$PATH"
PROMPT="$(cat "$ROOT/starter-prompt.txt") $(cat "$ROOT/tasks/$3")

$(cat "$ROOT/harness-note.txt")"
printf '%s\n' "$PROMPT" > "$T.prompt.txt"
cd "$T"
timeout 2700 claude -p "$PROMPT" --model "$2" --output-format stream-json --verbose \
  --dangerously-skip-permissions --setting-sources project,local --strict-mcp-config \
  > "$T.jsonl" 2> "$T.stderr.txt" || echo "exit $?" >> "$T.stderr.txt"
