"""Compact view of a claude stream-json transcript: each tool call, short results, text."""
import json
import sys

RESULT_CHARS = int(sys.argv[2]) if len(sys.argv) > 2 else 400


def short(s, n):
    s = s.replace("\r", "")
    return s if len(s) <= n else s[:n] + f" …[{len(s) - n} more]"


def main(path):
    with open(path, encoding="utf-8") as f:
        for line in f:
            try:
                ev = json.loads(line)
            except json.JSONDecodeError:
                continue
            t = ev.get("type")
            if t == "assistant":
                for c in ev["message"].get("content", []):
                    if c["type"] == "text" and c["text"].strip():
                        print("TEXT:", short(c["text"].strip(), 600))
                    elif c["type"] == "tool_use":
                        i = c["input"]
                        name = c["name"]
                        if name in ("Bash", "PowerShell"):
                            d = i.get("command", "")
                        elif name in ("Read", "Write", "Edit"):
                            d = i.get("file_path", "")
                            if name == "Read" and i.get("offset"):
                                d += f" @{i.get('offset')}+{i.get('limit')}"
                            if name == "Edit":
                                d += "\n    - " + short(i.get("old_string", ""), 200) + "\n    + " + short(i.get("new_string", ""), 200)
                        elif name in ("Grep", "Glob"):
                            d = f"{i.get('pattern')} in {i.get('path', '.')}"
                        elif name == "Skill":
                            d = i.get("skill", "")
                        else:
                            d = json.dumps(i)[:200]
                        print(f"CALL {name}: {short(d, 500)}")
            elif t == "user":
                content = ev["message"].get("content", [])
                if isinstance(content, str):
                    continue
                for c in content:
                    if c.get("type") == "tool_result":
                        r = c.get("content")
                        if isinstance(r, list):
                            r = " ".join(x.get("text", "<img>") if x.get("type") == "text" else "<image>" for x in r)
                        r = str(r)
                        flag = "ERR " if c.get("is_error") else ""
                        print(f"  -> {flag}{short(r, RESULT_CHARS)}")
            elif t == "result":
                print("RESULT:", ev.get("subtype"), "turns", ev.get("num_turns"), "cost", ev.get("total_cost_usd"), "ms", ev.get("duration_ms"))


main(sys.argv[1])
