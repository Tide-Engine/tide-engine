local singleton CheckObserved { int x = -1; int y = -1; int count = -1; }
view CheckState(GameState board, mut CheckObserved observed)
{
    Debug.Assert(board.x >= 0 && board.x < 5 && board.y >= 0 && board.y < 3);
    Debug.Assert(board.coins >= 0 && board.coins <= 2);
    if (observed.x != board.x || observed.y != board.y || observed.count != board.coins)
    {
        Debug.Log($"CHECK {board.x},{board.y},{board.coins}");
        observed.x = board.x; observed.y = board.y; observed.count = board.coins;
    }
}
