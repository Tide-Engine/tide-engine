settings { title = "Coin Collector"; tickRate = 30; maxPlayers = 1; }

// The board has 5 columns (x 0..4) and 3 rows (y 0..2); row 0 is the top.
// Cells are 64 world units wide and the board is centered on the origin, so
// column 0 is centered at x = -128 and row 0 at y = 64 (world Y points up).
float CellX(int column) { return -128 + column * 64; }
float CellY(int row) { return 64 - row * 64; }

// Where the player is, in cells. Each player's entity has an Owner.
component Player
{
    int x;
    int y;
}

// A coin stays in place; collecting it sets `collected` once and it never resets except on restart.
component Coin
{
    int x;
    int y;
    bool collected;
}

// Coins collected this round.
singleton Score { int collected; }

// Set during the tick a restart is requested, so coins can reset in the same tick.
singleton Round { bool restart; }

input Controls
{
    bool left;
    bool right;
    bool up;
    bool down;
    bool restart;

    Sample(Devices devices)
    {
        var keys = devices.keyboard;
        left = keys.leftArrow.pressed;
        right = keys.rightArrow.pressed;
        up = keys.upArrow.pressed;
        down = keys.downArrow.pressed;
        restart = keys.r.pressed;
    }
}

scene Main { }

event(Spawned) Setup(with Main)
{
    Spawn(Coin { x = 2, y = 1 });
    Spawn(Coin { x = 4, y = 2 });
}

event(PlayerJoined joined) AddPlayer()
{
    Spawn(Player { x = 0, y = 1 }, Owner { player = joined.player });
}

// Arrow presses use the input's edge (`.down`), so one press moves one cell.
system Steer(Controls input, mut Player player, mut Round round, mut Score score, with Owner)
{
    round.restart = input.restart.down;
    if (input.restart.down)
    {
        player.x = 0;
        player.y = 1;
        score.collected = 0;
    }
    else
    {
        if (input.left.down && player.x > 0) player.x -= 1;
        if (input.right.down && player.x < 4) player.x += 1;
        if (input.up.down && player.y > 0) player.y -= 1;
        if (input.down.down && player.y < 2) player.y += 1;
    }
}

system ResetCoins(mut Coin coin, Round round)
{
    if (round.restart) coin.collected = false;
}

// Runs after Steer (declared order), so the player has already moved this tick.
// `collected` is checked first, so a coin counts at most once.
system Collect(mut Coin coin, query (Player player, with Owner) players, mut Score score)
{
    foreach (var p in players)
    {
        if (!coin.collected && p.player.x == coin.x && p.player.y == coin.y)
        {
            coin.collected = true;
            score.collected += 1;
        }
    }
}

view DrawBoard(Score score, query (Coin coin) coins, query (Player player, with Owner) players)
{
    Draw.Clear(Color(0.09, 0.1, 0.12));
    Draw.SetCamera(float2(0, 0), Math.Max(180, 320 * Screen.height / Screen.width));

    for (var row = 0; row < 3; row++)
    {
        for (var column = 0; column < 5; column++)
        {
            var shade = (column + row) % 2 == 0 ? Color(0.2, 0.23, 0.27) : Color(0.16, 0.18, 0.21);
            Draw.Rect(float2(CellX(column), CellY(row)), float2(60, 60), shade);
        }
    }

    foreach (var coin in coins)
    {
        if (!coin.coin.collected)
            Draw.Circle(float2(CellX(coin.coin.x), CellY(coin.coin.y)), 16, Color.yellow);
    }

    foreach (var p in players)
    {
        Draw.Circle(float2(CellX(p.player.x), CellY(p.player.y)), 22, Color(0.35, 0.8, 1));
    }

    Draw.Text($"Coins: {score.collected} / 2", float2(-300, 170), 18, Color.white);
    Draw.Text("Arrows move, R restarts", float2(-300, -165), 14, Color.gray);
    if (score.collected == 2)
        Draw.Text("Complete! Press R to play again", float2(0, -140), 22, Color.green, Anchor.MiddleCenter);
}
