settings { title = "Pong"; tickRate = 60; maxPlayers = 2; }

const float HalfW = 320;
const float HalfH = 180;
const float PaddleX = 300;
const float PaddleHalfW = 5;
const float PaddleHalfH = 30;
const float BallR = 6;
const float PaddleSpeed = 280;
const float ServeSpeed = 220;
const int WinScore = 5;
const float WinSeconds = 3;

component Paddle
{
    int seat; // 0 = left, 1 = right
    float y;
}

singleton Game
{
    float2 ball;
    float2 vel;
    int left;
    int right;
    int serves;
    float serveDir = 1;
    int winner = -1; // -1 none, else seat of winner
    float winTimer;
}

input Controls
{
    float move;

    Sample(Devices devices)
    {
        var keys = devices.keyboard;
        if (keys.w.pressed || keys.upArrow.pressed) move += 1;
        if (keys.s.pressed || keys.downArrow.pressed) move -= 1;
    }
}

scene Main { }

event(PlayerJoined joined) AddPlayer(query (Paddle paddle) paddles)
{
    mut var leftTaken = false;
    foreach (var p in paddles)
        if (p.paddle.seat == 0) leftTaken = true;
    Spawn(Paddle { seat = leftTaken ? 1 : 0 }, Owner { player = joined.player });
}

event(PlayerLeft left) RemovePlayer(query (Owner owner) owners)
{
    foreach (var o in owners)
        if (o.owner.player == left.player) o.entity.Destroy();
}

system MovePaddle(Controls input, mut Paddle paddle, Time time)
{
    paddle.y = Math.Clamp(paddle.y + input.move * PaddleSpeed * time.dt, -HalfH + PaddleHalfH, HalfH - PaddleHalfH);
}

system Step(mut Game game, query (Paddle paddle) paddles, Time time)
{
    mut var count = 0;
    foreach (var p in paddles) count += 1;

    if (game.winner >= 0)
    {
        game.winTimer -= time.dt;
        if (game.winTimer <= 0)
        {
            game.winner = -1;
            game.left = 0;
            game.right = 0;
            game.serves = 0;
            game.ball = float2(0, 0);
            game.vel = float2(0, 0);
        }
        return;
    }

    if (count < 2)
    {
        game.ball = float2(0, 0);
        game.vel = float2(0, 0);
        return;
    }

    if (game.vel.x == 0)
    {
        // Serve from the center toward the player who conceded.
        var vy = (Math.Hash(game.serves) % 201 - 100) * 1.0;
        game.serves += 1;
        game.vel = float2(game.serveDir * ServeSpeed, vy);
        game.ball = float2(0, 0);
    }

    game.ball += game.vel * time.dt;

    if (game.ball.y > HalfH - BallR)
    {
        game.ball.y = HalfH - BallR;
        game.vel.y = -Math.Abs(game.vel.y);
    }
    if (game.ball.y < -HalfH + BallR)
    {
        game.ball.y = -HalfH + BallR;
        game.vel.y = Math.Abs(game.vel.y);
    }

    foreach (var p in paddles)
    {
        var dir = p.paddle.seat == 0 ? -1.0 : 1.0;
        var px = dir * PaddleX;
        var movingToward = game.vel.x * dir > 0;
        if (movingToward
            && Math.Abs(game.ball.x - px) < PaddleHalfW + BallR
            && Math.Abs(game.ball.y - p.paddle.y) < PaddleHalfH + BallR)
        {
            game.ball.x = px - dir * (PaddleHalfW + BallR);
            game.vel.x = -game.vel.x * 1.05;
            game.vel.y += (game.ball.y - p.paddle.y) * 4;
        }
    }

    if (game.ball.x < -HalfW - BallR)
    {
        game.right += 1;
        game.serveDir = -1;
        game.vel = float2(0, 0);
        game.ball = float2(0, 0);
    }
    else if (game.ball.x > HalfW + BallR)
    {
        game.left += 1;
        game.serveDir = 1;
        game.vel = float2(0, 0);
        game.ball = float2(0, 0);
    }

    if (game.left >= WinScore || game.right >= WinScore)
    {
        game.winner = game.left >= WinScore ? 0 : 1;
        game.winTimer = WinSeconds;
        game.vel = float2(0, 0);
        game.ball = float2(0, 0);
    }
}

view DrawCourt(Game game)
{
    Draw.SetCamera(float2(0, 0), HalfH);
    Draw.Clear(Color.black);
    for (var i = -8; i < 9; i++)
        Draw.Rect(float2(0, i * 20), float2(4, 10), Color.gray);
    Draw.Text($"{game.left}", float2(-60, 150), 48, Color.white, Anchor.MiddleCenter);
    Draw.Text($"{game.right}", float2(60, 150), 48, Color.white, Anchor.MiddleCenter);
    if (game.winner >= 0)
        Draw.Text(game.winner == 0 ? "Left player wins!" : "Right player wins!", float2(0, 0), 40, Color.yellow, Anchor.MiddleCenter);
    else
        Draw.Circle(game.ball, BallR, Color.white);
}

view DrawPaddles(Paddle paddle)
{
    var x = paddle.seat == 0 ? -PaddleX : PaddleX;
    Draw.Rect(float2(x, paddle.y), float2(PaddleHalfW * 2, PaddleHalfH * 2), Color.white);
}
