settings { title = "Credit Pool"; tickRate = 30; maxPlayers = 2; }

scene Main { }

component Wallet
{
    int credits = 3;
    int seat;
}

singleton Bank
{
    int pool;
    int nextSeat;
}

local singleton DepositUI { int amount = 1; }

command Deposit { int amount; }

event(PlayerJoined joined) AddWallet(mut Bank bank)
{
    Spawn(Wallet { seat = bank.nextSeat }, Owner { player = joined.player });
    bank.nextSeat += 1;
}

command(Deposit deposit, PlayerID sender) ApplyDeposit(Owner owner, mut Wallet wallet, mut Bank bank)
{
    if (sender != owner.player) return;
    if (deposit.amount < 1 || deposit.amount > 3) return;
    if (wallet.credits < deposit.amount) return;
    wallet.credits -= deposit.amount;
    bank.pool += deposit.amount;
}

float2 Seat(int seat) { return float2(seat == 0 ? 170 : 470, 170); }

view DrawBank(Bank bank)
{
    Draw.Clear(Color(0.035, 0.06, 0.09));
    Draw.SetScreen();
    Draw.RoundRect(float2(320, 75), float2(560, 100), 18, Color(0.08, 0.14, 0.18));
    Draw.Text("SHARED CREDIT POOL", float2(320, 35), 18, Color(0.55, 0.82, 0.78), Anchor.MiddleCenter);
    Draw.Text($"{bank.pool}", float2(320, 82), 52, Color.white, Anchor.MiddleCenter);
}

view DrawWallet(Wallet wallet, with Owner)
{
    var at = Seat(wallet.seat);
    var label = wallet.seat == 0 ? "PLAYER 1" : "PLAYER 2";
    Draw.Text($"{label}   {wallet.credits} credits", at, 20, Color(0.78, 0.87, 0.94), Anchor.MiddleCenter);
}

view DepositControls(mut DepositUI ui, with Wallet, with ownership)
{
    Draw.Text("YOUR DEPOSIT", float2(320, 218), 14, Color(0.55, 0.82, 0.78), Anchor.MiddleCenter);
    if (GUI.Button(Rect(185, 245, 70, 42), "−")) ui.amount = Math.Max(1, ui.amount - 1);
    Draw.Text($"{ui.amount}", float2(320, 266), 26, Color.white, Anchor.MiddleCenter);
    if (GUI.Button(Rect(385, 245, 70, 42), "+")) ui.amount = Math.Min(3, ui.amount + 1);
    if (GUI.Button(Rect(220, 305, 200, 42), $"DEPOSIT {ui.amount}")) this.Send(Deposit { amount = ui.amount });
}
