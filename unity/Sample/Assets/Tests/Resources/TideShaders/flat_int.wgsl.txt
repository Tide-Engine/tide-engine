struct Values { ids: vec4<i32> }
@group(0) @binding(0) var<uniform> u: Values;
struct Corner { @builtin(position) at: vec4<f32>, @location(0) @interpolate(flat) id: i32 }
@vertex fn vertex(@builtin(vertex_index) i: u32) -> Corner {
    var points = array<vec2<f32>, 3>(vec2<f32>(-1.0, -1.0), vec2<f32>(3.0, -1.0), vec2<f32>(-1.0, 3.0));
