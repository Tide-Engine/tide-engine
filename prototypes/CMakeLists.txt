# Prototypes have the same native/web game path as the public demos.
# Catalog games are staged at public addresses that can be shared.
# Prototypes here keep their opt-in build and shareable catalog staging.
if(TIDE_PRIVATE_PROTOTYPES)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cracked/CMakeLists.txt")
        add_subdirectory(cracked)
    endif()
    add_subdirectory(inside-out)
endif()
add_custom_target(private_games)
# Two Forks is built with the samples and staged at its prototype address.
if(TARGET sample_two_forks_page)
    add_dependencies(private_games sample_two_forks_page)
elseif(TARGET sample_two_forks)
    add_dependencies(private_games sample_two_forks)
endif()
if(TARGET cracked_page)
    add_dependencies(private_games cracked_page)
elseif(TARGET cracked)
    add_dependencies(private_games cracked)
endif()
if(TARGET inside_out_page)
    add_dependencies(private_games inside_out_page)
elseif(TARGET inside_out)
    add_dependencies(private_games inside_out)
endif()
